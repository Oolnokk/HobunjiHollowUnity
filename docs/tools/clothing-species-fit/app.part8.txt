async function composeRuntimePortrait(){
  const runtime=await runtimePortraitSet();
  const front=Object.assign(document.createElement('canvas'),{width:runtime.front.width,height:runtime.front.height});
  const back=Object.assign(document.createElement('canvas'),{width:runtime.back.width,height:runtime.back.height});
  const fctx=front.getContext('2d',{alpha:true}),bctx=back.getContext('2d',{alpha:true});
  fctx.drawImage(runtime.front,0,0);bctx.drawImage(runtime.back,0,0);

  if(E.showGarment.checked&&R.cloth){
    const warped=currentWarp();
    const layer=await rawGarmentToRuntimeLayer(warped,+E.opacity.value);
    if(layer){
      fctx.drawImage(layer,0,0);
      bctx.drawImage(flipCanvasX(layer),0,0);
    }
  }
  if(E.showAuthored.checked&&R.authoredTargetCloth){
    const authored=await rawGarmentToRuntimeLayer(R.authoredTargetCloth,+E.authoredOpacity.value);
    if(authored){
      fctx.drawImage(authored,0,0);
      bctx.drawImage(flipCanvasX(authored),0,0);
    }
  }
  return{...runtime,frontComposed:front,backComposed:back};
}
async function extendedRuntimeCanvas(){
  if(!R.extendedSource||!R.body)return null;
  const raw=currentExtendedWarp(false);
  if(!raw)return null;

  const normalSize=Number(window.SCRATCHBONES_CONFIG?.game?.assets?.pngPlaneAvatar?.previewPortraitCanvasSize)||200;
  const xf=targetBodyLayerXform();
  const normalRect=portraitLayerRect(R.body.width,R.body.height,xf,normalSize,normalSize);
  const pixelScaleY=normalRect.h/Math.max(1,R.body.height);
  const pixelScaleX=normalRect.w/Math.max(1,R.body.width);
  const drawW=raw.width*pixelScaleX;
  const drawH=raw.height*pixelScaleY;
  const drawX=normalRect.cx-drawW/2;
  const drawY=normalRect.y; // The imported raster shares the original sprite's TOP edge and only grows downward.
  const outHeight=Math.max(normalSize,Math.ceil(drawY+drawH)+2);

  const out=Object.assign(document.createElement('canvas'),{width:normalSize,height:outHeight});
  const ctx=out.getContext('2d',{alpha:true});ctx.imageSmoothingEnabled=false;
  ctx.drawImage(raw,drawX,drawY,drawW,drawH);
  await applyRuntimeCloudMask(out);
  return out;
}
function disposeObjectTree(root){
  if(!root)return;
  root.traverse?.(obj=>{
    if(obj.geometry?.dispose)obj.geometry.dispose();
    const materials=Array.isArray(obj.material)?obj.material:[obj.material].filter(Boolean);
    for(const material of materials){
      if(material?.map?.dispose)material.map.dispose();
      material?.dispose?.();
    }
  });
  root.parent?.remove?.(root);
}
