function flipCanvasX(source){
  const out=Object.assign(document.createElement('canvas'),{width:source.width,height:source.height});
  const ctx=out.getContext('2d',{alpha:true});
  ctx.translate(out.width,0);ctx.scale(-1,1);ctx.drawImage(source,0,0);
  return out;
}
async function composeRuntimePortrait(){
  const runtime=await runtimePortraitSet();
  const front=Object.assign(document.createElement('canvas'),{width:runtime.front.width,height:runtime.front.height});
  const back=Object.assign(document.createElement('canvas'),{width:runtime.back.width,height:runtime.back.height});
  const fctx=front.getContext('2d',{alpha:true}),bctx=back.getContext('2d',{alpha:true});
  fctx.drawImage(runtime.front,0,0);

  bctx.drawImage(runtime.back,0,0); // Canonical rear base body is always the backplane foundation.
  if(routedPatchIsBody()&&mainClothSource()&&E.showBody.checked){
    const warpedPatch=currentWarp();
    const patchLayer=await renderRoutedBodyPatchOverlay(
      runtime.profile,warpedPatch,R.patchWorkflow.descriptor,back.width,back.height
    );
    if(patchLayer)bctx.drawImage(patchLayer,0,0); // Explicit backplane order: patch is in front of the torso/body, while the clothing block below still draws afterward and therefore remains in front of the patch.
  }

  if(mainClothSource()&&!routedPatchIsBody()&&E.showGarment.checked){
    const warped=currentWarp();
    const layer=await rawGarmentToRuntimeLayer(warped,+E.opacity.value);
    if(layer){
      fctx.drawImage(layer,0,0);
      bctx.drawImage(flipCanvasX(layer),0,0);
    }
  }
  if(E.showAuthored.checked&&R.authoredTargetCloth){
    const authored=await rawGarmentToRuntimeLayer(R.authoredTargetCloth,+E.authoredOpacity.value);
    if(authored){
      fctx.drawImage(authored,0,0);
      bctx.drawImage(flipCanvasX(authored),0,0);
    }
  }
  return{...runtime,frontComposed:front,backComposed:back};
}
async function extendedRuntimeCanvas(){
  if(!R.extendedSource||!R.body)return null;
  const raw=currentExtendedWarp(false);
  if(!raw)return null;

  const normalSize=Number(window.SCRATCHBONES_CONFIG?.game?.assets?.pngPlaneAvatar?.previewPortraitCanvasSize)||200;
  const xf=targetBodyLayerXform();
  const normalRect=portraitLayerRect(R.body.width,R.body.height,xf,normalSize,normalSize);
  const pixelScaleY=normalRect.h/Math.max(1,R.body.height);
  const pixelScaleX=normalRect.w/Math.max(1,R.body.width);
  const drawW=raw.width*pixelScaleX;
  const drawH=raw.height*pixelScaleY;
  const drawX=normalRect.cx-drawW/2;
  const drawY=normalRect.y; // The imported raster shares the original sprite's TOP edge and only grows downward.
  const outHeight=Math.max(normalSize,Math.ceil(drawY+drawH)+2);

  const out=Object.assign(document.createElement('canvas'),{width:normalSize,height:outHeight});
  const ctx=out.getContext('2d',{alpha:true});ctx.imageSmoothingEnabled=false;
  ctx.drawImage(raw,drawX,drawY,drawW,drawH);
  await applyRuntimeCloudMask(out);
  return out;
}
function disposeObjectTree(root){
  if(!root)return;
  root.traverse?.(obj=>{
    if(obj.geometry?.dispose)obj.geometry.dispose();
    const materials=Array.isArray(obj.material)?obj.material:[obj.material].filter(Boolean);
    for(const material of materials){
      if(material?.map?.dispose)material.map.dispose();
      material?.dispose?.();
    }
  });
  root.parent?.remove?.(root);
}
function disposeRuntimeRig(){
  try{R.three.feet?.dispose?.()}catch{}

  if(R.three.avatarModel&&window.PNGPlaneAvatar?.disposeAvatarModel){
    try{window.PNGPlaneAvatar.disposeAvatarModel(R.three.avatarModel)}catch{}
  }
  if(R.three.rigRoot)disposeObjectTree(R.three.rigRoot);
  R.three.rigRoot=null;R.three.avatarModel=null;R.three.extendedGroup=null;
}
function ensureThreePreview(){
  if(R.three.ready)return true;
  if(!window.THREE){E.threeStatus.textContent='Three.js failed to load; 2D tool still works.';return false}
  const THREE=window.THREE;
  const scene=new THREE.Scene();
  scene.background=new THREE.Color(0x05080b);
  scene.add(new THREE.HemisphereLight(0xffffff,0x24303b,1.2));
  const directional=new THREE.DirectionalLight(0xffffff,1.1);directional.position.set(2.5,4,3);scene.add(directional);

  const camera=new THREE.PerspectiveCamera(38,1,0.01,100);
  camera.position.set(0,.55,R.three.zoom);
  const renderer=new THREE.WebGLRenderer({antialias:true,alpha:false,preserveDrawingBuffer:true});
  renderer.setPixelRatio(Math.min(2,window.devicePixelRatio||1));
  renderer.setClearColor(0x05080b,1);
  E.threeHost.innerHTML='';E.threeHost.appendChild(renderer.domElement);

  const group=new THREE.Group();group.name='ClothingFitRuntimeViewRoot';scene.add(group);
  const grid=new THREE.GridHelper(4,16,0x4b6175,0x263440);grid.position.y=0;scene.add(grid);

  R.three.scene=scene;R.three.camera=camera;R.three.renderer=renderer;R.three.group=group;R.three.ready=true;

  function resize(){
    const w=Math.max(1,E.threeHost.clientWidth),h=Math.max(1,E.threeHost.clientHeight);
    renderer.setSize(w,h,false);camera.aspect=w/h;camera.updateProjectionMatrix();
  }
  new ResizeObserver(resize).observe(E.threeHost);resize();

  renderer.domElement.addEventListener('pointerdown',e=>{
    R.three.drag={x:e.clientX,y:e.clientY,rotX:R.three.rotX,rotY:R.three.rotY};
    renderer.domElement.setPointerCapture(e.pointerId);e.preventDefault();
  });
  renderer.domElement.addEventListener('pointermove',e=>{
    if(!R.three.drag)return;
    R.three.rotY=R.three.drag.rotY+(e.clientX-R.three.drag.x)*0.008;
    R.three.rotX=Math.max(-1.0,Math.min(1.0,R.three.drag.rotX+(e.clientY-R.three.drag.y)*0.008));
    group.rotation.set(R.three.rotX,R.three.rotY,0);e.preventDefault();
  });
  const end=e=>{R.three.drag=null;if(e?.pointerId!=null&&renderer.domElement.hasPointerCapture?.(e.pointerId))renderer.domElement.releasePointerCapture(e.pointerId)};
  renderer.domElement.addEventListener('pointerup',end);renderer.domElement.addEventListener('pointercancel',end);
  renderer.domElement.addEventListener('wheel',e=>{
    R.three.zoom=Math.max(1.1,Math.min(8,R.three.zoom+Math.sign(e.deltaY)*0.2));
    camera.position.z=R.three.zoom;e.preventDefault();
  },{passive:false});
  (function loop(){requestAnimationFrame(loop);renderer.render(scene,camera)})();
  E.threeStatus.textContent='3D scene ready; loading runtime rig…';
  return true;
}
function frameRuntimeRig(){
  if(!R.three.rigRoot||!window.THREE)return;
  const THREE=window.THREE,box=new THREE.Box3().setFromObject(R.three.rigRoot);
  if(box.isEmpty())return;
  const center=box.getCenter(new THREE.Vector3()),size=box.getSize(new THREE.Vector3());
  const fov=THREE.MathUtils.degToRad(R.three.camera.fov||38);
  const aspect=Math.max(.1,R.three.camera.aspect||1);
  const horizontalFov=2*Math.atan(Math.tan(fov/2)*aspect);
  const byH=(size.y/2+.18)/Math.max(.01,Math.tan(fov/2));
  const byW=(size.x/2+.18)/Math.max(.01,Math.tan(horizontalFov/2));
  const distance=Math.max(1.25,byH,byW)+size.z/2;
