function canonicalPortraitImageForTint(img,sourceKey,tint){
  const resolver=window._imageForTint; // portrait-utils' own image+tint resolver; using it preserves the exact body-color algorithm and cache behavior.
  if(typeof resolver!=='function')throw new Error('Canonical portrait tint resolver is unavailable; cannot safely inject a body patch without changing avatar colors.');
  return resolver(img,sourceKey,tint);
}
function runtimeBodyLayerRecord(profile,descriptor=null){
  const layers=profile?.fighter?.bodyLayers||[]; // Exact runtime-normalized body-layer records; these carry the same preset/transform fields portrait-utils will use for the real avatar.
  const wanted=String(descriptor?.layerId||descriptor?.role||'').toLowerCase();
  if(wanted){
    const exact=layers.find(layer=>String(layer?.id||'').toLowerCase()===wanted);
    if(exact)return exact;
    const normalizedWanted=wanted.replace(/[^a-z0-9]/g,'');
    const normalized=layers.find(layer=>String(layer?.id||'').toLowerCase().replace(/[^a-z0-9]/g,'')===normalizedWanted);
    if(normalized)return normalized;
  }
  return layers.find(layer=>String(layer?.id||'').toLowerCase()==='torso')||layers[0]||null;
}
function patchCanvasDataUrl(canvas){
  try{return canvas?.toDataURL?.('image/png')||''}catch(error){throw new Error(`Could not encode routed body patch for portrait assembly: ${error?.message||error}`)}
}
async function renderRoutedBodyPatchOverlay(profile,rawPatchCanvas,descriptor,width,height){
  if(!profile||!rawPatchCanvas)return null;
  const runtimeLayer=runtimeBodyLayerRecord(profile,descriptor);
  if(!runtimeLayer)throw new Error('Could not resolve the selected body layer inside the runtime portrait profile.');
  const patchUrl=patchCanvasDataUrl(rawPatchCanvas);
  if(!patchUrl)throw new Error('Could not encode the routed body patch as an isolated portrait layer.');

  const overlay=Object.assign(document.createElement('canvas'),{width,height}); // Transparent rear-body overlay; drawn after the ordinary rear body and before clothing in composeRuntimePortrait().
  const isolatedLayer={...runtimeLayer,url:patchUrl}; // Reuses the exact runtime body-layer transform/preset, tint slot, position, and any future metadata while swapping only its raster.
  const isolatedFighter={
    ...profile.fighter,
    id:`__clothing_fit_body_patch_${safeName(E.species.value)}_${safeName(E.gender.value)}_${safeName(isolatedLayer.id||'layer')}`, // Unique id prevents resolvePortraitFighter() from replacing this one-layer fighter with the canonical full-body fighter.
    headUrl:null,headFallbackUrls:[],urLayers:[],bodyLayers:[isolatedLayer]
  };
  const isolatedProfile={...profile,fighter:isolatedFighter};
  await withPreviewCloudMaskTransform(()=>window.NpcAvatarPreview.renderProfileToCanvas(
    overlay,isolatedProfile,{
      seatId:'clothing-fit-runtime-body-patch',portraitView:'behind',forceEyesOpen:true,
      omitHeadSpriteAndCosmetics:true
    }
  )); // portrait-utils owns tinting, preset resolution, rear flip, breathing/body deformation, and mask placement; no transform is reconstructed here.
  return overlay;
}
function bodyPatchImageForTintResolver(rawPatchCanvas,descriptor){
  const targetLayer=targetBodyLayerRecord(descriptor);
  const targetUrl=targetLayer?.url||descriptor?.targetUrl||'';
  const compositeByImage=new WeakMap(); // Reuses the same pre-tint composite if the portrait renderer asks for the target layer more than once in one rear render.
  const routeRevision=R.patchWorkflow.revision; // Makes canonical tint caches distinguish this live patch state from earlier replacements/drags.
  return function bodyPatchAwareImageForTint(img,sourceKey,tint){
    if(!samePortraitAsset(sourceKey,targetUrl))return canonicalPortraitImageForTint(img,sourceKey,tint);
    let composite=compositeByImage.get(img);
    if(!composite){
      const width=Number(img?.naturalWidth||img?.width)||rawPatchCanvas.width;
      const height=Number(img?.naturalHeight||img?.height)||rawPatchCanvas.height;
      composite=Object.assign(document.createElement('canvas'),{width,height});
      const ctx=composite.getContext('2d',{alpha:true});ctx.imageSmoothingEnabled=false;
      ctx.drawImage(img,0,0,width,height); // The ordinary body layer remains the base raster; the corrective patch becomes part of this same layer rather than a second avatar plane.
      ctx.drawImage(rawPatchCanvas,0,0,rawPatchCanvas.width,rawPatchCanvas.height,0,0,width,height); // Full-sprite coordinates are preserved; only a dimension mismatch is proportionally normalized to the actual target raster.
      compositeByImage.set(img,composite);
    }
    const patchedSourceKey=`${targetUrl}#hobunji-body-patch-${routeRevision}-${currentWarpCacheSignature()}`;
    return canonicalPortraitImageForTint(composite,patchedSourceKey,tint); // Tint the original body pixels and patch together before the renderer applies its own transform preset, rear flip, breathing/deformation, and mask.
  };
}
function currentWarpCacheSignature(){
  const tr=currentTransform();
  const offsets=currentOffsets();
  return `${mode}:${tr.scaleX},${tr.scaleY},${tr.translateX},${tr.translateY}:${offsets.map(p=>`${Number(p.x).toFixed(4)},${Number(p.y).toFixed(4)}`).join(';')}`;
}
function portraitLayerRect(rawWidth,rawBaseHeight,xform,targetWidth=200,targetHeight=200){
  const L=80;
  const h=L*xform.sy;
  const w=(rawWidth/Math.max(1,rawBaseHeight))*L*Math.abs(xform.sx);
  const cx=targetWidth/2+xform.ay*L;
  const cy=targetHeight/2-xform.ax*L;
  return{x:cx-w/2,y:cy-h/2,w,h,cx,cy};
}
function previewedCloudMaskXform(rawXform={}){
  const baseXform=normalizedLayerXform(rawXform);
  const preview=currentMaskTransform();
  return{
    ...baseXform,
    sx:(Number(baseXform.sx)||1)*(Number(preview.scaleX)||1),
    scaleX:(Number(baseXform.sx)||1)*(Number(preview.scaleX)||1),
    sy:(Number(baseXform.sy)||1)*(Number(preview.scaleY)||1),
    scaleY:(Number(baseXform.sy)||1)*(Number(preview.scaleY)||1),
    // portrait-utils uses cy = centerY - ax*L, so subtracting preview Y makes
    // positive slider values move the mask downward like the other Y controls.
    ax:(Number(baseXform.ax)||0)-(Number(preview.translateY)||0),
  };
}
async function withPreviewCloudMaskTransform(callback){
  const original=window.applyPortraitOpacityMask;
  if(typeof original!=='function')return callback();
  const wrapped=function previewCloudMaskTransform(ctx,image,xform){
    return original(ctx,image,previewedCloudMaskXform(xform||{}));
  };
  // Preserve the runtime marker because arm-cloud-mask checks it to avoid
  // installing another canonical Y-scale wrapper around the temporary function.
  if(original.__hobunjiCloudMaskYScaled)wrapped.__hobunjiCloudMaskYScaled=true;
  window.applyPortraitOpacityMask=wrapped;
  try{
    return await callback();
  }finally{
    window.applyPortraitOpacityMask=original;
  }
}

async function applyRuntimeCloudMask(canvas){
  const cfg=R.speciesCfg.get(E.species.value)?.[E.gender.value];
  const maskLayer=cfg?.portraitOpacityMaskLayer;
  if(!maskLayer?.url||typeof window.applyPortraitOpacityMask!=='function')return canvas;
  const image=await loadImage(assetUrl(maskLayer.url));
  if(!image)return canvas;
  const ctx=canvas.getContext('2d',{alpha:true});
  window.applyPortraitOpacityMask(ctx,image,previewedCloudMaskXform(maskLayer.xform||{}));
  return canvas;
}
