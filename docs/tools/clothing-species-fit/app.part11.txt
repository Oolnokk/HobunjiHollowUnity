
  const targetKeyText=readLine('target grid');
  const identity=targetKeyText.match(/^(.+)_(male|female)$/);
  if(!identity)throw new Error('Could not read the target species and gender from “target grid”.');
  const [,species,gender]=identity;
  const garment=readLine('garment');
  if(!garment||garment==='(none)')throw new Error('The pasted report does not identify a garment.');

  let transform;
  try{transform=JSON.parse(readLine('base transform'))}catch{throw new Error('Could not parse the base transform JSON.');}
  for(const key of ['scaleX','scaleY','translateX','translateY']){
    if(!Number.isFinite(Number(transform?.[key])))throw new Error(`The base transform is missing a valid ${key}.`);
  }

  const section=text.split('source→target points:')[1]?.split('recent log:')[0];
  if(!section)throw new Error('The report is missing its source→target point list.');
  const points=new Array(GRID_SIZE*GRID_SIZE);
  const pointPattern=/^\s*(\d+):\s*\(([-+\d.eE]+),([-+\d.eE]+)\)\s*->\s*\(([-+\d.eE]+),([-+\d.eE]+)\)\s*$/gm;
  let match;
  while((match=pointPattern.exec(section))){
    const index=Number(match[1]),sx=Number(match[2]),sy=Number(match[3]),x=Number(match[4]),y=Number(match[5]);
    if(index<0||index>=points.length||points[index])throw new Error(`Point ${index} is duplicated or outside the 6×6 grid.`);
    const expectedX=(index%GRID_SIZE)/(GRID_SIZE-1),expectedY=Math.floor(index/GRID_SIZE)/(GRID_SIZE-1);
    if(![sx,sy,x,y].every(Number.isFinite)||Math.abs(sx-expectedX)>.002||Math.abs(sy-expectedY)>.002)
      throw new Error(`Point ${index} is invalid or does not match the 6×6 source grid.`);
    if(Math.abs(x)>10||Math.abs(y)>10)throw new Error(`Point ${index} is outside the supported coordinate range.`);
    points[index]={x,y};
  }
  const receivedPointCount=points.filter(Boolean).length;
  if(receivedPointCount!==points.length)throw new Error(`The report contains ${receivedPointCount} of 36 required grid points.`);

  let maskTransform=null;
  const maskLine=readLine('cloud mask preview');
  if(maskLine){try{maskTransform=JSON.parse(maskLine)}catch{}}
  return{species,gender,garment,transform,points,maskTransform};
}

async function importGrids(file){
  const v=JSON.parse(await file.text());
  if(v.gridSize!==GRID_SIZE)throw new Error('This file is not a 6×6 species-grid export.');
  pushUndo();
  if(v.version===3||v.version===2){
    A.baseTransforms=v.baseTransforms||{};A.localOffsets=v.localOffsets||{};A.maskPreviewTransforms=v.maskPreviewTransforms||{};
  }else if(v.version===1&&v.targetGrids){
    A.baseTransforms={};A.localOffsets={};A.maskPreviewTransforms={};
    for(const [key,grid] of Object.entries(v.targetGrids)){
      if(!Array.isArray(grid)||grid.length!==SOURCE_GRID.length)continue;
      A.baseTransforms[key]=defaultTransform();
      A.localOffsets[key]=grid.map((p,i)=>({x:p.x-SOURCE_GRID[i].x,y:p.y-SOURCE_GRID[i].y}));
      A.maskPreviewTransforms[key]=defaultMaskTransform();
    }
  }else throw new Error('Unsupported grid export version.');
  if(v.repoBase)E.repo.value=v.repoBase;
  saveState();syncTransformControls();syncMaskTransformControls();await refreshAssets();log('Imported '+file.name);
}

function collectPatchImageNodes(node,path=[],out=[]){
  if(!node||typeof node!=='object')return out;
  if(node.image?.url)out.push({path:[...path],url:node.image.url,role:node.layerRole||'',drawSlot:node.drawSlot||'',attachBone:node.attachBone||''});
  for(const [key,value] of Object.entries(node)){
    if(key==='image')continue;
    if(value&&typeof value==='object')collectPatchImageNodes(value,[...path,key],out);
  }
  return out;
}
function inferPatchView(path,url){

}
