async function importGrids(file){
  const v=JSON.parse(await file.text());
  if(v.gridSize!==GRID_SIZE)throw new Error('This file is not a 6×6 species-grid export.');
  pushUndo();
  if(v.version===3||v.version===2){
    A.baseTransforms=v.baseTransforms||{};A.localOffsets=v.localOffsets||{};A.maskPreviewTransforms=v.maskPreviewTransforms||{};
  }else if(v.version===1&&v.targetGrids){
    A.baseTransforms={};A.localOffsets={};A.maskPreviewTransforms={};
    for(const [key,grid] of Object.entries(v.targetGrids)){
      if(!Array.isArray(grid)||grid.length!==SOURCE_GRID.length)continue;
      A.baseTransforms[key]=defaultTransform();
      A.localOffsets[key]=grid.map((p,i)=>({x:p.x-SOURCE_GRID[i].x,y:p.y-SOURCE_GRID[i].y}));
      A.maskPreviewTransforms[key]=defaultMaskTransform();
    }
  }else throw new Error('Unsupported grid export version.');
  if(v.repoBase)E.repo.value=v.repoBase;
  saveState();syncTransformControls();syncMaskTransformControls();await refreshAssets();log('Imported '+file.name);
}
