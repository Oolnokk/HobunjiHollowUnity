function assetUrl(path){

  if(p.startsWith('./assets/'))return docsUrl(p.slice(2));

function buildDerivedOverlayLayer(layer,cutoffY,normalHeight){
  const clipped=document.createElement('canvas');clipped.width=layer.width;clipped.height=layer.height;
  const cx=clipped.getContext('2d',{alpha:true});cx.imageSmoothingEnabled=false;
  cx.drawImage(layer.img,0,0);
  cx.clearRect(0,0,clipped.width,Math.max(0,Math.min(layer.height,cutoffY)));
  const shiftY=(normalHeight-cutoffY);
  const out=document.createElement('canvas');
  out.width=layer.width;out.height=Math.max(normalHeight,layer.height+shiftY);
  const ox=out.getContext('2d',{alpha:true});ox.imageSmoothingEnabled=false;
  ox.drawImage(clipped,0,shiftY);
  return{
    img:out,name:layer.name,width:out.width,height:out.height,
    _sourceHeight:layer.height,_cutoffY:cutoffY,_alignShiftY:shiftY,_derivedFromRegularSprite:true
  };
}
function rebuildExtendedSource(){
  const imported=importedExtendedLayers();
  if(!imported.length){R.extendedSource=null;R.extendedWarped=null;updateExtendedStatus();return null}
  const width=R.maoBody?.width||imported[0].width;
  const normalHeight=R.maoBody?.height||0;
  const cutoffY=syncExtendedCutoffControls();
  const cutoffMode=extendedSourceUsesCutoffMode();
  const layers=cutoffMode?imported.map(layer=>buildDerivedOverlayLayer(layer,cutoffY,normalHeight)):imported;

  const height=Math.max(normalHeight,...layers.map(l=>l.height));
  const canvas=document.createElement('canvas');canvas.width=width;canvas.height=height;
  const x=canvas.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;
  const mismatches=[];
  for(const layer of layers){
    if(layer.width!==width)mismatches.push(`${layer.name}: ${layer.width}px wide (portrait is ${width}px)`);
    x.drawImage(layer.img,0,0);
  }
  canvas._paths=layers.map(l=>l.name);
  canvas._overlayMode=cutoffMode?'derived-cutoff':'top-aligned';
  canvas._cutoffY=cutoffMode?cutoffY:null;
  canvas._alignShiftY=cutoffMode?(normalHeight-cutoffY):0;
  canvas._sourceLayerHeights=layers.map(l=>l.height);
  R.extendedSource=canvas;R.extendedWarped=null;
  updateExtendedStatus(mismatches);
  return canvas;
}
function updateExtendedStatus(mismatches=[]){
  const roles=[['torso',R.extendedLayers.torso],['torso clothing',R.extendedLayers.torsoClothing],['overwear',R.extendedLayers.overwear]];
  const lines=roles.map(([role,v])=>`${role}: ${v?`${v.name} (${v.width}×${v.height})`:'—'}`);
  const cutoffY=syncExtendedCutoffControls();
  lines.push(`source mode: ${extendedSourceUsesCutoffMode()?'derive overlay from regular sprite':'direct extended import, top-aligned'}`);
  lines.push(`cutoff Y: ${cutoffY}`);
  if(R.extendedSource){
    if(R.extendedSource._overlayMode==='derived-cutoff'){
      lines.push(`composite: ${R.extendedSource.width}×${R.extendedSource.height}, cutoff-aligned, shifted by ${R.extendedSource._alignShiftY}px so cutoff row meets portrait bottom`);
    }else{
      lines.push(`composite: ${R.extendedSource.width}×${R.extendedSource.height}, top-aligned`);
    }
  }
  if(mismatches.length)lines.push('WIDTH WARNING: '+mismatches.join(' | '));
  lines.push('arms: skipped');
  E.extendedStatus.textContent=lines.join('\\n');
}
