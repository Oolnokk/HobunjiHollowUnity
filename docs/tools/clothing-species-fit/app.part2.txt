  if(R.images.has(url))return R.images.get(url);
  const p=new Promise((ok,bad)=>{
    const img=new Image();img.crossOrigin='anonymous';img.onload=()=>ok(img);img.onerror=()=>bad(new Error('Image failed: '+url));img.src=url;
  });
  R.images.set(url,p);
  p.catch(()=>{if(R.images.get(url)===p)R.images.delete(url)}); // A newly added sprite can succeed on a later refresh rather than reusing an old rejected image.
  return p;
}
function loadLocalImage(file){
  return new Promise((resolve,reject)=>{
    if(!file){resolve(null);return}
    const url=URL.createObjectURL(file),img=new Image();
    img.onload=()=>{URL.revokeObjectURL(url);resolve({img,name:file.name,width:img.naturalWidth,height:img.naturalHeight})};
    img.onerror=()=>{URL.revokeObjectURL(url);reject(new Error('Could not load '+file.name))};
    img.src=url;
  }); // Local extended sprites never leave the browser.
}

function rebuildExtendedSource(){
  const layers=importedExtendedLayers();
  if(!layers.length){R.extendedSource=null;R.extendedWarped=null;updateExtendedStatus();return null}
  const width=R.sourceBody?.width||layers[0].width;
  const normalHeight=R.sourceBody?.height||0;
  const height=Math.max(normalHeight,...layers.map(l=>l.height));
  const canvas=document.createElement('canvas');canvas.width=width;canvas.height=height;
  const x=canvas.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;
  const mismatches=[];
  for(const layer of layers){
    if(layer.width!==width)mismatches.push(`${layer.name}: ${layer.width}px wide (portrait is ${width}px)`);
    x.drawImage(layer.img,0,0); // Top-left anchored, no resizing; expected extended sprites share portrait width.
  }
  canvas._paths=layers.map(l=>l.name);
  R.extendedSource=canvas;R.extendedWarped=null;
  updateExtendedStatus(mismatches);
  return canvas;
}
function updateExtendedStatus(mismatches=[]){
  const roles=[
    ['torso',R.extendedLayers.torso],
    ['torso clothing',R.extendedLayers.torsoClothing],
    ['overwear',R.extendedLayers.overwear]
  ];
  const lines=roles.map(([role,v])=>`${role}: ${v?`${v.name} (${v.width}×${v.height})`:'—'}`);
  if(R.extendedSource)lines.push(`composite: ${R.extendedSource.width}×${R.extendedSource.height}, top-aligned`);
  if(mismatches.length)lines.push('WIDTH WARNING: '+mismatches.join(' | '));
  lines.push('arms: skipped');
  E.extendedStatus.textContent=lines.join('\n');
}
async function mapLimit(items,n,fn){
  const out=new Array(items.length);let q=0;
  async function run(){while(true){const i=q++;if(i>=items.length)return;try{out[i]=await fn(items[i],i)}catch(error){out[i]={error}}}}
  await Promise.all(Array.from({length:Math.min(n,items.length)},run));return out;
}

async function loadRepositoryData(){
  status('Loading repository data…');
  R.three.portraitCache.clear();
  if(R.three.runtimeDocsBase&&R.three.runtimeDocsBase!==runtimeDocsBase()){
    R.three.runtimeReadyPromise=null;
    R.three.runtimeDocsBase=null;
    E.threeStatus.textContent='Repository base changed; 3D runtime will reload from the new root.';
  }
  R.speciesCfg.clear();R.garments=[];R.images.clear();R.bodyCanvases.clear();R.garmentCanvases.clear();
  try{
    const [si,ci]=await Promise.all([fetchJson('config/species/index.json'),fetchJson('config/cosmetics/index.json')]);
    R.speciesIndex=si.entries||[];

    const speciesResults=await mapLimit(R.speciesIndex,6,async entry=>({
      entry,cfg:await fetchJson('config/species/'+String(entry.path||'').replace(/^\.\//,''))
    }));
    for(const x of speciesResults)if(x&&!x.error)R.speciesCfg.set(x.entry.speciesId,x.cfg);

    const cosmeticResults=await mapLimit(ci.entries||[],8,async entry=>({
      entry,cfg:await fetchJson('config/cosmetics/'+String(entry.path||'').replace(/^\.\//,''))
    }));
    for(const x of cosmeticResults){
      if(!x||x.error)continue;
      const cfg=x.cfg,variants=cfg?.speciesVariants||{};
      if((BODY_SLOTS.has(cfg?.slot)||HEAD_SLOTS.has(cfg?.slot))&&
         (Object.keys(variants).length||(HEAD_SLOTS.has(cfg?.slot)&&cfg?.parts?.head))){
        R.garments.push({id:x.entry.id,label:cfg?.meta?.name||x.entry.id,slot:cfg.slot,cfg,
          configPath:`config/cosmetics/${String(x.entry.path||'').replace(/^\.\//,'')}`}); // Ready-to-install ZIPs can include updated target config when needed.
      }
    }
    R.garments.sort((a,b)=>a.label.localeCompare(b.label));
    rebuildSelectors();await refreshAssets();
    status(`Loaded ${R.speciesCfg.size} species and ${R.garments.length} body/headwear items. Starting sprites: ${E.sourceSpecies.value} ${E.sourceGender.value}.`);
    log('Repository load complete.');
  }catch(error){status('Load failed: '+error.message);log('ERROR '+error.message);console.error(error)}
}

function garmentRecord(){return R.garments.find(g=>g.id===E.garment.value)||null}
function headwearLayerRecords(garment,species,gender){
  const cfg=garment?.cfg,variant=cfg?.speciesVariants?.[`${species}_${gender}`];
  const parts=variant?.parts||(cfg?.onlyExactSpeciesVariant?null:cfg?.parts);
  const layers=parts?.head?.layers||{};
  return Object.entries(layers).filter(([,layer])=>layer?.image?.url&&
    !['exposed','exposedSkin'].includes(String(layer.layerRole||''))).map(([name,layer])=>({
      name,layer,path:layer.image.url,isBack:/^(back|rear|behind)/i.test(name),
      behindPath:layer.behindImage?.url||null // A hat can use different back-view art without moving its front piece behind the skull.
    })); // A hood's rear cloth is not interchangeable with front cloth or exposed-species ear pixels.
}
async function buildHeadReferenceCanvas(species,gender,width,height){
  const cfg=R.speciesCfg.get(species)?.[gender],url=cfg?.headSprite;
  const canvas=Object.assign(document.createElement('canvas'),{width,height});
  if(!url){
    log(`No head sprite configured for ${species} ${gender}; using a blank reference until a sprite is authored.`);
    canvas._paths=[];return canvas;
  }
  let image;
  try{image=await loadImage(assetUrl(url))}catch(error){
    log(`Head reference not available for ${species} ${gender}: ${error.message}`);
    canvas._paths=[url];return canvas;
  }
  const ctx=canvas.getContext('2d',{alpha:true});ctx.imageSmoothingEnabled=false;
  const sourceHead=window.getPortraitXformPreset?.('B')||{ax:-.0983,ay:-.0809,sx:2.49,sy:2.49};
  const bodyXform=normalizedLayerXform(R.speciesCfg.get(species)?.[gender]?.portraitBodyLayers?.find(layer=>layer.id==='torso')?.xform||{ax:-.2,ay:0,sx:2.55,sy:2.55});
  const headRect=portraitLayerRect(image.naturalWidth,image.naturalHeight,sourceHead);
  const bodyRect=portraitLayerRect(width,height,bodyXform);
  const scaleX=width/bodyRect.w,scaleY=height/bodyRect.h;
  ctx.drawImage(image,(headRect.x-bodyRect.x)*scaleX,(headRect.y-bodyRect.y)*scaleY,headRect.w*scaleX,headRect.h*scaleY);
  canvas._paths=[url];return canvas; // Inverse of canonical portrait draw positions, so head and headwear share the raw clothing-stage coordinate space.
}
async function buildVariantGarmentCanvas(id,species,gender,width,height){
  const key=`${id}::${species}::${gender}::${width}x${height}::${fitRegion()}`;
  if(R.garmentCanvases.has(key))return R.garmentCanvases.get(key);
  const p=(async()=>{
    const garment=R.garments.find(x=>x.id===id),variant=garment?.cfg?.speciesVariants?.[`${species}_${gender}`];
    if(HEAD_SLOTS.has(garment?.slot)){
      const records=headwearLayerRecords(garment,species,gender);
      if(!records.length)return null;
      const front=Object.assign(document.createElement('canvas'),{width,height});
      const back=Object.assign(document.createElement('canvas'),{width,height});
      const behindView=Object.assign(document.createElement('canvas'),{width,height});
      let hasFront=false,hasBack=false,hasBehindOverride=false; // Avoid wasting headwear warps and exports on unused orientations.
      const exportedLayers=[]; // Keeps headwear materials/front-back source identities separate for the fitted ZIP.
      for(const record of records){
        try{
          const image=await loadImage(assetUrl(record.path));
          const target=record.isBack?back:front;
          const ctx=target.getContext('2d',{alpha:true});ctx.imageSmoothingEnabled=false;
          ctx.drawImage(image,0,0); // All authored headwear layers retain their source-pixel alignment.
          const layerCanvas=Object.assign(document.createElement('canvas'),{width,height});
          layerCanvas.getContext('2d',{alpha:true}).drawImage(image,0,0);
          const layerPath=['parts','head','layers',record.name]; // Matches the same config leaf used by the behind-patch author.
          exportedLayers.push({name:`head.layers.${record.name}`,path:layerPath,url:record.path,role:record.layer.layerRole||'',view:record.isBack?'under':'front',canvas:layerCanvas});
          if(record.isBack)hasBack=true;
          else{
            hasFront=true;
            let rearImage=image;
            if(record.behindPath&&record.behindPath!==record.path){
              try{
                rearImage=await loadImage(assetUrl(record.behindPath));
                const rearCanvas=Object.assign(document.createElement('canvas'),{width,height}); // Independent rear-view override, never substituted for the front layer.
                rearCanvas.getContext('2d',{alpha:true}).drawImage(rearImage,0,0);
                exportedLayers.push({name:`head.layers.${record.name}.behindImage`,path:[...layerPath,'behindImage'],url:record.behindPath,role:record.layer.layerRole||'',view:'behind',canvas:rearCanvas});
              }catch(error){log(`Missing rear ${garment.label} layer ${record.behindPath}; falling back to front art: ${error.message}`)}
            }
            const rearCtx=behindView.getContext('2d',{alpha:true});rearCtx.imageSmoothingEnabled=false;
            rearCtx.drawImage(rearImage,0,0); // Compose cap then metal band in the same author-specified layer order.
            if(record.behindPath)hasBehindOverride=true;
          }
        }catch(error){log(`Missing ${garment.label} layer ${record.path}: ${error.message}`)}
      }
      const canvas=Object.assign(document.createElement('canvas'),{width,height});
      const x=canvas.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;
      x.drawImage(back,0,0);x.drawImage(front,0,0);
      canvas._front=hasFront?front:null;canvas._back=hasBack?back:null;
      canvas._behindView=hasBehindOverride?behindView:null; // Dedicated rear view is distinct from cloth behind the skull.
      canvas._layers=exportedLayers; // Independent fitted outputs use identical cage transforms but preserve the original config layer topology.
      canvas._paths=exportedLayers.map(layer=>layer.url);
      return canvas;
    }
    if(!variant)return null; // Missing authored target species/gender body art is normal.
    const layers=collectPatchImageNodes(variant); // One leaf per authored PNG, not one per unique URL or flattened composite.
    if(!layers.length)return null;
    const images=await Promise.all(layers.map(layer=>loadImage(assetUrl(layer.url))));
    const canvas=document.createElement('canvas');canvas.width=width;canvas.height=height;
    const x=canvas.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;
    const exportedLayers=[]; // Body garments can contain independently colored/positioned nested parts.
