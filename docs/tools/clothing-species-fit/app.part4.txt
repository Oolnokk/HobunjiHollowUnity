  const bottomDest=[];
  const extraDepth=Math.max(0,sourceHeight-sourceNormalHeight);
  const sourceCellHeight=sourceNormalHeight/(GRID_SIZE-1);
  const extensionFactor=sourceCellHeight>0?extraDepth/sourceCellHeight:0;
  for(let col=0;col<GRID_SIZE;col++){
    const prev=targetPx[(GRID_SIZE-2)*GRID_SIZE+col];
    const last=targetPx[(GRID_SIZE-1)*GRID_SIZE+col];
    bottomDest.push({
      x:last.x+(last.x-prev.x)*extensionFactor,
      y:last.y+(last.y-prev.y)*extensionFactor
    });
  }
  const maxY=Math.max(normalOutHeight,...targetPx.map(p=>p.y),...bottomDest.map(p=>p.y));
  const outHeight=Math.max(normalOutHeight,Math.ceil(maxY)+2);
  const output=document.createElement('canvas');output.width=outWidth;output.height=outHeight;
  const x=output.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;

  // Normal portrait-height region: source cage occupies only the first normal portrait-height pixels.
  for(let row=0;row<GRID_SIZE-1;row++){
    for(let col=0;col<GRID_SIZE-1;col++){
      const tl=row*GRID_SIZE+col,tr=tl+1,bl=(row+1)*GRID_SIZE+col,br=bl+1;
      const sTL={x:SOURCE_GRID[tl].x*sourceWidth,y:SOURCE_GRID[tl].y*sourceNormalHeight};
      const sTR={x:SOURCE_GRID[tr].x*sourceWidth,y:SOURCE_GRID[tr].y*sourceNormalHeight};
      const sBL={x:SOURCE_GRID[bl].x*sourceWidth,y:SOURCE_GRID[bl].y*sourceNormalHeight};
      const sBR={x:SOURCE_GRID[br].x*sourceWidth,y:SOURCE_GRID[br].y*sourceNormalHeight};
      drawBilinearCell(x,source,sTL,sTR,sBL,sBR,targetPx[tl],targetPx[tr],targetPx[bl],targetPx[br]);
    }
  }

  // Extra vertical region: continue the deformation vector from the last normal grid row.
  if(sourceHeight>sourceNormalHeight){
    const extSub=Math.max(CELL_SUBDIVISIONS,Math.ceil(Math.max(1,extensionFactor)*CELL_SUBDIVISIONS));
    for(let col=0;col<GRID_SIZE-1;col++){
      const lastL=targetPx[(GRID_SIZE-1)*GRID_SIZE+col],lastR=targetPx[(GRID_SIZE-1)*GRID_SIZE+col+1];
      const sTL={x:(col/(GRID_SIZE-1))*sourceWidth,y:sourceNormalHeight};
      const sTR={x:((col+1)/(GRID_SIZE-1))*sourceWidth,y:sourceNormalHeight};
      const sBL={x:(col/(GRID_SIZE-1))*sourceWidth,y:sourceHeight};
      const sBR={x:((col+1)/(GRID_SIZE-1))*sourceWidth,y:sourceHeight};
      drawBilinearCell(x,source,sTL,sTR,sBL,sBR,lastL,lastR,bottomDest[col],bottomDest[col+1],extSub);
    }
  }
  return output;
}
function warpGarment(source,targetNorm,outWidth,outHeight){
  const output=document.createElement('canvas');output.width=outWidth;output.height=outHeight;
  const x=output.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;
  if(source.width===outWidth&&source.height===outHeight&&
     targetNorm.every((point,index)=>Math.abs(point.x-SOURCE_GRID[index].x)<1e-9&&Math.abs(point.y-SOURCE_GRID[index].y)<1e-9)){
    x.drawImage(source,0,0);return output; // An untouched grid is an identity map; copy pixels instead of resampling 800 triangles.
  }

  /*
    Important correction from V4:
    a 6x6 square was formerly split into only two triangles. A corner node therefore
    had zero influence over the opposite half of that same square. Here each base cell
    is treated as a bilinear quad. We sample that bilinear surface into a 4x4 micro-grid,
    then render the micro-quads as affine triangles. Thus every one of the four authored
    corner nodes influences every pixel region in its adjacent cell.
  */
  for(let row=0;row<GRID_SIZE-1;row++){
    for(let col=0;col<GRID_SIZE-1;col++){
      const tl=row*GRID_SIZE+col,tr=tl+1,bl=(row+1)*GRID_SIZE+col,br=bl+1;

      const sTL=SOURCE_GRID[tl],sTR=SOURCE_GRID[tr],sBL=SOURCE_GRID[bl],sBR=SOURCE_GRID[br];
      const dTL=targetNorm[tl],dTR=targetNorm[tr],dBL=targetNorm[bl],dBR=targetNorm[br];

      for(let sy=0;sy<CELL_SUBDIVISIONS;sy++){
        const v0=sy/CELL_SUBDIVISIONS,v1=(sy+1)/CELL_SUBDIVISIONS;
        for(let sx=0;sx<CELL_SUBDIVISIONS;sx++){
          const u0=sx/CELL_SUBDIVISIONS,u1=(sx+1)/CELL_SUBDIVISIONS;

          const sn00=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u0,v0),source.width,source.height);
          const sn10=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u1,v0),source.width,source.height);
          const sn01=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u0,v1),source.width,source.height);
          const sn11=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u1,v1),source.width,source.height);

          const dn00=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u0,v0),outWidth,outHeight);
          const dn10=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u1,v0),outWidth,outHeight);
          const dn01=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u0,v1),outWidth,outHeight);
          const dn11=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u1,v1),outWidth,outHeight);

          // Alternate the micro-cell diagonal to avoid reintroducing a strong one-direction triangle bias.
          if(((row+col+sx+sy)&1)===0){
            drawTriangleWarp(x,source,sn00,sn10,sn11,dn00,dn10,dn11);
            drawTriangleWarp(x,source,sn00,sn11,sn01,dn00,dn11,dn01);
          }else{
            drawTriangleWarp(x,source,sn00,sn10,sn01,dn00,dn10,dn01);
            drawTriangleWarp(x,source,sn10,sn11,sn01,dn10,dn11,dn01);
          }
        }
      }
    }
  }
  return output;
}

async function buildSourceReferenceCanvas(species,gender,garmentId,targetBody){
  if(bodyLayers(species,gender).length)return buildBodyCanvas(species,gender);
  const garment=R.garments.find(record=>record.id===garmentId);

    ?headwearLayerRecords(garment,species,gender).flatMap(layer=>[layer.path,layer.behindPath])
    :collectPatchImageNodes(garment?.cfg?.speciesVariants?.[`${species}_${gender}`]).map(layer=>layer.url);
  const urls=[...new Set(paths.filter(Boolean))]; // Collect every material and alternate rear raster before sizing the shared source.

    const fallback=document.createElement('canvas');fallback.width=targetBody.width;fallback.height=targetBody.height;
    fallback._paths=[];return fallback;

  const images=loaded.filter(result=>result.status==='fulfilled').map(result=>result.value);
  if(!images.length)throw new Error(`Could not load any ${species} ${gender} garment sprites to establish source dimensions.`);
  const fallback=document.createElement('canvas');

  fallback.height=Math.max(...images.map(image=>image.naturalHeight||image.height));

  log(`Using ${species} ${gender} garment sprite dimensions (${fallback.width}×${fallback.height}) without body reference sprites.${missing?` ${missing} source images unavailable.`:''}`);
  return fallback;
} // NPC-only or unimplemented body rigs can still supply valid starting clothing artwork.

function drawGrid(ctx,points,color,active){
  const px=toDisplayPixels(points,E.stage.width,E.stage.height);
  ctx.save();ctx.strokeStyle=color;ctx.fillStyle=color;ctx.globalAlpha=active?1:.58;ctx.lineWidth=1.2;

  for(let row=0;row<GRID_SIZE;row++){
    ctx.beginPath();
    for(let col=0;col<GRID_SIZE;col++){
      const p=px[row*GRID_SIZE+col];
      col?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y);
    }
    ctx.stroke();
  }
  for(let col=0;col<GRID_SIZE;col++){
    ctx.beginPath();
    for(let row=0;row<GRID_SIZE;row++){
      const p=px[row*GRID_SIZE+col];
      row?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y);
    }
    ctx.stroke();
  }
  for(const p of px){ctx.beginPath();ctx.arc(p.x,p.y,HANDLE_RADIUS,0,Math.PI*2);ctx.fill()}
  ctx.restore();
}

function applyHeadbandSplineWarp(garment,state,width,height){
  if(!state?.enabled||!window.ClothingHeadbandSpline.valid(state.source)||!window.ClothingHeadbandSpline.valid(state.target))return garment;
  if(state.source.every((point,index)=>Math.abs(point.x-state.target[index].x)<1e-6&&Math.abs(point.y-state.target[index].y)<1e-6))
    return garment; // An unshaped curve is an identity map; skip the resample so the 6×6 fit exports untouched pixels.
  const ctx=garment.getContext('2d',{alpha:true});
  const rgba=ctx.getImageData(0,0,width,height);
  const warped=window.ClothingHeadbandSpline.warp(rgba.data,width,height,state.source,state.target);
  const result=document.createElement('canvas');result.width=width;result.height=height;
  const targetCtx=result.getContext('2d',{alpha:true}),image=targetCtx.createImageData(width,height);
  image.data.set(warped);targetCtx.putImageData(image,0,0);
  return result; // PNG alpha and black outlines remain authored pixels, not regenerated outlines.
}

function detectHoodOpening(canvas){
  const Cut=window.PortraitBeardHoodCutout,Hole=window.ClothingHoodHoleDeform;
  if(!Cut||!Hole||!canvas)return null;
  const width=canvas.width,height=canvas.height,rect=holePortraitRect(width,height);
  const portrait=Object.assign(document.createElement('canvas'),{width:200,height:200});
  const pctx=portrait.getContext('2d',{alpha:true});pctx.imageSmoothingEnabled=false;
  pctx.drawImage(canvas,rect.x,rect.y,rect.w,rect.h);
  const rgba200=pctx.getImageData(0,0,200,200).data;
  const lower=Cut.lowerHoodOpeningSegment(rgba200),upper=Cut.upperHoodOpeningSegment(rgba200);
  const ring=Hole.ringFromPortrait(lower,upper,rect,width,height);
  if(!ring)return null;
  const rgba=canvas.getContext('2d').getImageData(0,0,width,height).data;
  const profile=Hole.buildProfile(rgba,width,height,ring.points,ring.seed);
  if(profile){profile.ringPoints=ring.points;profile.clockwiseComplete=!!lower.clockwiseComplete}
  return profile;
}
function holeProfileForGrid(grid,width,height){

}

// Where the fit (6×6 cage, then the hood-hole deform for base cloth) moves a source pixel.
function forwardFitPoint(p,grid,width,height,hole){
  const N=GRID_SIZE-1,u=Math.max(0,Math.min(1,p.x/width)),v=Math.max(0,Math.min(1,p.y/height));
  const col=Math.min(N-1,Math.floor(u*N)),row=Math.min(N-1,Math.floor(v*N));
  const tl=row*GRID_SIZE+col,tr=tl+1,bl=tl+GRID_SIZE,br=bl+1;
  const q=bilerpPoint(grid[tl],grid[tr],grid[bl],grid[br],u*N-col,v*N-row);
  let x=q.x*width,y=q.y*height;
  const Hole=window.ClothingHoodHoleDeform;
  if(Hole&&hole?.enabled&&!Hole.isIdentity(hole.corners)){
    const profile=holeProfileForGrid(grid,width,height);
    if(profile){const d=Hole.displacementAt(profile,Hole.normalizeCorners(hole.corners),Hole.normalizeRate(hole.rate),x,y,0);x+=d.x;y+=d.y}
  }
  return{x,y};
}
function applySilhouettePins(canvas,grid,width,height,pins,hole){
  const Pins=window.ClothingSilhouettePins;
  if(!Pins||!canvas||!pins?.points?.length)return canvas;
  const list=pins.points.map(p=>{const f=forwardFitPoint(p,grid,width,height,hole);return{x:f.x,y:f.y,ex:p.x-f.x,ey:p.y-f.y}});
  const ctx=canvas.getContext('2d',{alpha:true}),image=ctx.getImageData(0,0,width,height);
  const warped=Pins.warp(image.data,width,height,list,pins.sigma);
  if(warped===image.data)return canvas;
  const result=Object.assign(document.createElement('canvas'),{width,height});
  const rctx=result.getContext('2d',{alpha:true}),out=rctx.createImageData(width,height);
  out.data.set(warped);rctx.putImageData(out,0,0);
  return result;
}
// How closely a layer rides along with the opening: trim follows it (so the
// base can show where the trim moved away), every other material uses the
// cloth falloff. Older saved records have no trimFollow and follow fully.
function holeFollowForRole(hole,role){
  if(String(role||'')!=='trim')return 0;
  const value=Number(hole?.trimFollow);
  return Number.isFinite(value)?Math.max(0,Math.min(1,value)):1;
}
function applyHoleDeform(canvas,grid,width,height,hole,role=''){
  const Hole=window.ClothingHoodHoleDeform;
  if(!Hole||!hole?.enabled||Hole.isIdentity(hole.corners))return canvas;
  const profile=holeProfileForGrid(grid,width,height);
  if(!profile)return canvas;
  // Forward mesh warp: the region inside the (smoothed) hood outline is
  // erased and redrawn through displaced polar triangles, the same affine
  // triangle drawing the 6×6 grid uses. Every layer goes through the same
  // mesh, so base, trim and any other material move together, and growing or
  // shrinking the opening stretches the cloth instead of tearing it.
  const mesh=Hole.mesh(profile,hole.corners,hole.rate,{follow:holeFollowForRole(hole,role)});
  if(!mesh)return canvas;
  const result=Object.assign(document.createElement('canvas'),{width,height});
  const rctx=result.getContext('2d',{alpha:true});rctx.imageSmoothingEnabled=false;
  rctx.drawImage(canvas,0,0);
  const edge=mesh.rings[mesh.rings.length-1]; // Pad ring: d === s, outside every moving pixel.
  rctx.save();rctx.globalCompositeOperation='destination-out';rctx.beginPath();
  edge.forEach((v,i)=>i?rctx.lineTo(v.s.x,v.s.y):rctx.moveTo(v.s.x,v.s.y));
  rctx.closePath();rctx.fill();rctx.restore();
  Hole.forEachTriangle(mesh,(s0,s1,s2,d0,d1,d2)=>drawTriangleWarp(rctx,canvas,s0,s1,s2,d0,d1,d2));
  return result;
}
function warpClothing(source,grid,width,height,fitContourState=null,options={}){
  const isHeadwear=fitRegion()==='headwear';
  const state=fitContourState||(isHeadwear?currentHeadbandRecord():currentContourRecord());
  const garment=isHeadwear&&state?.enabled
    ?cachedHeadbandGridWarp(source,grid,width,height)
    :warpGarment(source,grid,width,height);
  if(isHeadwear){
    const splined=applyHeadbandSplineWarp(garment,state,width,height);
    if(options.view==='behind'){ // Rear-view art has no face opening; only its own pinned outline applies.
      const pins=options.pins!==undefined?options.pins:currentPinRecord();
      return applySilhouettePins(splined,grid,width,height,pins?.rearPoints?.length?{points:pins.rearPoints,sigma:pins.sigma}:null,null);
    }
    const hole=options.hole!==undefined?options.hole:currentHoleRecord();
    const holed=applyHoleDeform(splined,grid,width,height,hole,options.role);
    return applySilhouettePins(holed,grid,width,height,options.pins!==undefined?options.pins:currentPinRecord(),hole); // Pinned outline returns to its source position last.
  }
  if(routedPatchIsBody()||!R.sourceTorso||!R.targetTorso)return garment;
  if(!state?.enabled)return garment;
  const mask=fitContourState?.maskPixels||currentContourMask();
  const start=warpedSourceTorso(grid,width,height);
  if(!start||R.targetTorso.width!==width||R.targetTorso.height!==height)return garment;
  const data=garment.getContext('2d').getImageData(0,0,width,height);
  const donor=start.getContext('2d').getImageData(0,0,width,height);
  const target=R.targetTorso.getContext('2d').getImageData(0,0,width,height);
  const result=window.ClothingContourFit.fitPixels(data.data,donor.data,target.data,mask,width,height,state);
  const canvas=document.createElement('canvas');canvas.width=width;canvas.height=height;
  const context=canvas.getContext('2d',{alpha:true}),pixels=context.createImageData(width,height);
  pixels.data.set(result.pixels);context.putImageData(pixels,0,0); // Uses the existing browser canvas API on mobile without needing a global ImageData constructor.
  R.contourLastFit={rows:result.rows,limitedRows:result.limitedRows}; // Mobile diagnostics show when an extreme row needed contraction backoff.
  return canvas;
}
function warpGridForMode(){return mode==='transform'?transformedSourceGrid():targetGrid()}
function layerMatchesRoutedClothingPatch(layer){
  const descriptor=R.patchWorkflow.descriptor;
  return R.patchWorkflow.active&&descriptor?.kind==='clothing'&&
    Array.isArray(descriptor.layerPath)&&JSON.stringify(layer?.path)===JSON.stringify(descriptor.layerPath);
} // A corrective image belongs to one exact cosmetic config leaf, not the entire garment.

function fittedSourceCanvasForLayer(layer){
  if(!layerMatchesRoutedClothingPatch(layer)||!R.patchWorkflow.canvas)return layer.canvas;
  if(R.patchWorkflow._patchedLayer===layer&&R.patchWorkflow._patchedLayerCanvas)return R.patchWorkflow._patchedLayerCanvas;
  const canvas=document.createElement('canvas'); // Fits the patched native raster in the garment's shared source-pixel frame.
  canvas.width=R.cloth.width;canvas.height=R.cloth.height;
  const context=canvas.getContext('2d',{alpha:true});context.imageSmoothingEnabled=false;
  context.drawImage(R.patchWorkflow.canvas,0,0); // Do not stretch a smaller sprite to the full portrait.
  R.patchWorkflow._patchedLayer=layer;R.patchWorkflow._patchedLayerCanvas=canvas;
  return canvas;
}

function mainClothSource(){
  if(!R.patchWorkflow.active||!R.patchWorkflow.canvas)return R.cloth;
  if(R.patchWorkflow.descriptor?.kind!=='clothing'||!R.cloth?._layers?.length)return R.patchWorkflow.canvas; // Body patches remain patch-only.
  if(R.patchWorkflow._compositeBase===R.cloth&&R.patchWorkflow._compositeCanvas)return R.patchWorkflow._compositeCanvas;
  const layers=R.cloth._layers,matched=layers.some(layer=>layerMatchesRoutedClothingPatch(layer));
  if(!matched){
    if(!R.patchWorkflow._warnedMissingLayer){log('Patch source no longer matches a garment layer; using the unpatched garment.');R.patchWorkflow._warnedMissingLayer=true}
    return R.cloth;
  }
  const canvas=document.createElement('canvas');canvas.width=R.cloth.width;canvas.height=R.cloth.height;
  const context=canvas.getContext('2d',{alpha:true});context.imageSmoothingEnabled=false;
  for(const layer of layers)context.drawImage(fittedSourceCanvasForLayer(layer),0,0); // Keep every other material, in authored draw order.
  R.patchWorkflow._compositeBase=R.cloth;R.patchWorkflow._compositeCanvas=canvas; // Reuse during frequent 2D and 3D redraws.
  return canvas;
