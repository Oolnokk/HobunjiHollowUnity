  const bottomDest=[];
  const extraDepth=Math.max(0,sourceHeight-sourceNormalHeight);
  const sourceCellHeight=sourceNormalHeight/(GRID_SIZE-1);
  const extensionFactor=sourceCellHeight>0?extraDepth/sourceCellHeight:0;
  for(let col=0;col<GRID_SIZE;col++){
    const prev=targetPx[(GRID_SIZE-2)*GRID_SIZE+col];
    const last=targetPx[(GRID_SIZE-1)*GRID_SIZE+col];
    bottomDest.push({
      x:last.x+(last.x-prev.x)*extensionFactor,
      y:last.y+(last.y-prev.y)*extensionFactor
    });
  }
  const maxY=Math.max(normalOutHeight,...targetPx.map(p=>p.y),...bottomDest.map(p=>p.y));
  const outHeight=Math.max(normalOutHeight,Math.ceil(maxY)+2);
  const output=document.createElement('canvas');output.width=outWidth;output.height=outHeight;
  const x=output.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;

  // Normal portrait-height region: source cage occupies only the first normal portrait-height pixels.
  for(let row=0;row<GRID_SIZE-1;row++){
    for(let col=0;col<GRID_SIZE-1;col++){
      const tl=row*GRID_SIZE+col,tr=tl+1,bl=(row+1)*GRID_SIZE+col,br=bl+1;
      const sTL={x:SOURCE_GRID[tl].x*sourceWidth,y:SOURCE_GRID[tl].y*sourceNormalHeight};
      const sTR={x:SOURCE_GRID[tr].x*sourceWidth,y:SOURCE_GRID[tr].y*sourceNormalHeight};
      const sBL={x:SOURCE_GRID[bl].x*sourceWidth,y:SOURCE_GRID[bl].y*sourceNormalHeight};
      const sBR={x:SOURCE_GRID[br].x*sourceWidth,y:SOURCE_GRID[br].y*sourceNormalHeight};
      drawBilinearCell(x,source,sTL,sTR,sBL,sBR,targetPx[tl],targetPx[tr],targetPx[bl],targetPx[br]);
    }
  }

  // Extra vertical region: continue the deformation vector from the last normal grid row.
  if(sourceHeight>sourceNormalHeight){
    const extSub=Math.max(CELL_SUBDIVISIONS,Math.ceil(Math.max(1,extensionFactor)*CELL_SUBDIVISIONS));
    for(let col=0;col<GRID_SIZE-1;col++){
      const lastL=targetPx[(GRID_SIZE-1)*GRID_SIZE+col],lastR=targetPx[(GRID_SIZE-1)*GRID_SIZE+col+1];
      const sTL={x:(col/(GRID_SIZE-1))*sourceWidth,y:sourceNormalHeight};
      const sTR={x:((col+1)/(GRID_SIZE-1))*sourceWidth,y:sourceNormalHeight};
      const sBL={x:(col/(GRID_SIZE-1))*sourceWidth,y:sourceHeight};
      const sBR={x:((col+1)/(GRID_SIZE-1))*sourceWidth,y:sourceHeight};
      drawBilinearCell(x,source,sTL,sTR,sBL,sBR,lastL,lastR,bottomDest[col],bottomDest[col+1],extSub);
    }
  }
  return output;
}
function warpGarment(source,targetNorm,outWidth,outHeight){
  const output=document.createElement('canvas');output.width=outWidth;output.height=outHeight;
  const x=output.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;

  /*
    Important correction from V4:
    a 6x6 square was formerly split into only two triangles. A corner node therefore
    had zero influence over the opposite half of that same square. Here each base cell
    is treated as a bilinear quad. We sample that bilinear surface into a 4x4 micro-grid,
    then render the micro-quads as affine triangles. Thus every one of the four authored
    corner nodes influences every pixel region in its adjacent cell.
  */
  for(let row=0;row<GRID_SIZE-1;row++){
    for(let col=0;col<GRID_SIZE-1;col++){
      const tl=row*GRID_SIZE+col,tr=tl+1,bl=(row+1)*GRID_SIZE+col,br=bl+1;

      const sTL=SOURCE_GRID[tl],sTR=SOURCE_GRID[tr],sBL=SOURCE_GRID[bl],sBR=SOURCE_GRID[br];
      const dTL=targetNorm[tl],dTR=targetNorm[tr],dBL=targetNorm[bl],dBR=targetNorm[br];

      for(let sy=0;sy<CELL_SUBDIVISIONS;sy++){
        const v0=sy/CELL_SUBDIVISIONS,v1=(sy+1)/CELL_SUBDIVISIONS;
        for(let sx=0;sx<CELL_SUBDIVISIONS;sx++){
          const u0=sx/CELL_SUBDIVISIONS,u1=(sx+1)/CELL_SUBDIVISIONS;

          const sn00=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u0,v0),source.width,source.height);
          const sn10=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u1,v0),source.width,source.height);
          const sn01=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u0,v1),source.width,source.height);
          const sn11=toWarpPixel(bilerpPoint(sTL,sTR,sBL,sBR,u1,v1),source.width,source.height);

          const dn00=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u0,v0),outWidth,outHeight);
          const dn10=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u1,v0),outWidth,outHeight);
          const dn01=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u0,v1),outWidth,outHeight);
          const dn11=toWarpPixel(bilerpPoint(dTL,dTR,dBL,dBR,u1,v1),outWidth,outHeight);

          // Alternate the micro-cell diagonal to avoid reintroducing a strong one-direction triangle bias.
          if(((row+col+sx+sy)&1)===0){
            drawTriangleWarp(x,source,sn00,sn10,sn11,dn00,dn10,dn11);
            drawTriangleWarp(x,source,sn00,sn11,sn01,dn00,dn11,dn01);
          }else{
            drawTriangleWarp(x,source,sn00,sn10,sn01,dn00,dn10,dn01);
            drawTriangleWarp(x,source,sn10,sn11,sn01,dn10,dn11,dn01);
          }
        }
      }
    }
  }
  return output;
}

async function buildSourceReferenceCanvas(species,gender,garmentId,targetBody){
  if(bodyLayers(species,gender).length)return buildBodyCanvas(species,gender);
  const garment=R.garments.find(record=>record.id===garmentId);
  const firstUrl=HEAD_SLOTS.has(garment?.slot)
    ?headwearLayerRecords(garment,species,gender)[0]?.path
    :collectPatchImageNodes(garment?.cfg?.speciesVariants?.[`${species}_${gender}`])[0]?.url;
  if(!firstUrl){
    log(`No ${species} ${gender} source body or garment raster; retaining target dimensions as an empty fitting reference.`);
    const fallback=document.createElement('canvas');fallback.width=targetBody.width;fallback.height=targetBody.height;
    fallback._paths=[];return fallback;
  }
  const image=await loadImage(assetUrl(firstUrl)); // The garment's native source dimensions replace missing species-body dimensions.
  const fallback=document.createElement('canvas');
  fallback.width=image.naturalWidth||image.width;fallback.height=image.naturalHeight||image.height;
  fallback._paths=[firstUrl];
  log(`Using ${species} ${gender} garment raster dimensions because that species has no body reference sprites.`);
  return fallback;
} // NPC-only or unimplemented body rigs can still supply valid starting clothing artwork.

async function refreshAssets(){
  const generation=++R.assetGeneration; // The latest selection alone may install downloaded images.
  const species=E.species.value,gender=E.gender.value,garmentId=E.garment.value,region=fitRegion();
  const sourceSpecies=E.sourceSpecies.value,sourceGender=E.sourceGender.value; // Source dimensions and garment raster are independent of the target avatar.
  R.body=null;R.sourceBody=null;R.cloth=null;R.headReference=null;R.authoredTargetCloth=null;
  E.clothPng.disabled=true;E.compositePng.disabled=true; // Never allow exporting a stale garment during a source/target change.
  try{
    if(R.patchWorkflow.active||R.patchWorkflow.canvas){
      R.patchWorkflow={active:false,canvas:null,descriptor:null,filename:'',patchName:'',revision:(R.patchWorkflow.revision||0)+1};
      E.clothPng.textContent=region==='headwear'?'Export fitted headwear PNG':'Export fitted clothing PNG';
    }
    const body=await buildBodyCanvas(species,gender);
    const sourceBody=await buildSourceReferenceCanvas(sourceSpecies,sourceGender,garmentId,body); // Body art is optional for source-only NPC species.
    const headReference=region==='headwear'?await buildHeadReferenceCanvas(species,gender,body.width,body.height):null;
    const cloth=garmentId?await buildGarmentCanvas(garmentId,sourceSpecies,sourceGender,sourceBody.width,sourceBody.height):null;
    const authored=garmentId
      ?await buildVariantGarmentCanvas(garmentId,species,gender,body.width,body.height)
      :null;
    if(generation!==R.assetGeneration)return; // Switching modes/species while images load cannot reinstall stale art.
    R.body=body;R.sourceBody=sourceBody;R.headReference=headReference;R.cloth=cloth;R.authoredTargetCloth=authored;
    const layerCount=cloth?._layers?.length||0; // Distinct authored source rasters, including explicit alternate behind-view images.
    E.clothPng.disabled=!cloth;E.compositePng.disabled=!cloth;
    E.clothPng.textContent=layerCount>1?`Export ${layerCount} fitted layers ZIP`:(region==='headwear'?'Export fitted headwear PNG':'Export fitted clothing PNG');
    E.sourceLayerStatus.textContent=cloth
      ?`${sourceSpecies} ${sourceGender}: ${layerCount} authored source layer${layerCount===1?'':'s'} → ${species} ${gender}. ${layerCount>1?'Export preserves every layer in a ZIP.':'Single-layer PNG export.'}`
      :`No source garment art for ${sourceSpecies} ${sourceGender} in this slot. Choose another item or starting sprite.`;
    E.exportHeadwearFront.disabled=region!=='headwear'||!cloth?._front;
    E.exportHeadwearBack.disabled=region!=='headwear'||!(cloth?._behindView||cloth?._back);
    E.authoredStatus.textContent=authored
      ?`Authored ${species}_${gender} variant available.`
      :'No authored target variant for this garment/species/gender.';
    ensureTargetData(species,gender);syncTransformControls();syncMaskTransformControls();
    saveState();render();
    window.dispatchEvent(new CustomEvent('hobunji-clothing-fit-assets-refreshed',{detail:{species,gender,garmentId}}));
  }catch(error){
    if(generation!==R.assetGeneration)return;
    E.sourceLayerStatus.textContent='Source/target reload failed: '+error.message;
    status('Refresh failed: '+error.message);log('ERROR '+error.message);console.error(error);
  }
}

function drawGrid(ctx,points,color,active){
  const px=toDisplayPixels(points,E.stage.width,E.stage.height);
  ctx.save();ctx.strokeStyle=color;ctx.fillStyle=color;ctx.globalAlpha=active?1:.58;ctx.lineWidth=1.2;

  for(let row=0;row<GRID_SIZE;row++){
    ctx.beginPath();
    for(let col=0;col<GRID_SIZE;col++){
      const p=px[row*GRID_SIZE+col];
      col?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y);
    }
    ctx.stroke();
  }
  for(let col=0;col<GRID_SIZE;col++){
    ctx.beginPath();
    for(let row=0;row<GRID_SIZE;row++){
      const p=px[row*GRID_SIZE+col];
      row?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y);
    }
    ctx.stroke();
  }
  for(const p of px){ctx.beginPath();ctx.arc(p.x,p.y,HANDLE_RADIUS,0,Math.PI*2);ctx.fill()}
  ctx.restore();
}

function layerMatchesRoutedClothingPatch(layer){
  const descriptor=R.patchWorkflow.descriptor;
  return R.patchWorkflow.active&&descriptor?.kind==='clothing'&&
    Array.isArray(descriptor.layerPath)&&JSON.stringify(layer?.path)===JSON.stringify(descriptor.layerPath);
} // A corrective image belongs to one exact cosmetic config leaf, not the entire garment.

function fittedSourceCanvasForLayer(layer){
  if(!layerMatchesRoutedClothingPatch(layer)||!R.patchWorkflow.canvas)return layer.canvas;
  if(R.patchWorkflow._patchedLayer===layer&&R.patchWorkflow._patchedLayerCanvas)return R.patchWorkflow._patchedLayerCanvas;
  const canvas=document.createElement('canvas'); // Fits the patched native raster in the garment's shared source-pixel frame.
  canvas.width=R.cloth.width;canvas.height=R.cloth.height;
  const context=canvas.getContext('2d',{alpha:true});context.imageSmoothingEnabled=false;
  context.drawImage(R.patchWorkflow.canvas,0,0); // Do not stretch a smaller sprite to the full portrait.
  R.patchWorkflow._patchedLayer=layer;R.patchWorkflow._patchedLayerCanvas=canvas;
  return canvas;
}

function mainClothSource(){
  if(!R.patchWorkflow.active||!R.patchWorkflow.canvas)return R.cloth;
  if(R.patchWorkflow.descriptor?.kind!=='clothing'||!R.cloth?._layers?.length)return R.patchWorkflow.canvas; // Body patches remain patch-only.
  if(R.patchWorkflow._compositeBase===R.cloth&&R.patchWorkflow._compositeCanvas)return R.patchWorkflow._compositeCanvas;
  const layers=R.cloth._layers,matched=layers.some(layer=>layerMatchesRoutedClothingPatch(layer));
  if(!matched){
    if(!R.patchWorkflow._warnedMissingLayer){log('Patch source no longer matches a garment layer; using the unpatched garment.');R.patchWorkflow._warnedMissingLayer=true}
    return R.cloth;
  }
  const canvas=document.createElement('canvas');canvas.width=R.cloth.width;canvas.height=R.cloth.height;
  const context=canvas.getContext('2d',{alpha:true});context.imageSmoothingEnabled=false;
  for(const layer of layers)context.drawImage(fittedSourceCanvasForLayer(layer),0,0); // Keep every other material, in authored draw order.
  R.patchWorkflow._compositeBase=R.cloth;R.patchWorkflow._compositeCanvas=canvas; // Reuse during frequent 2D and 3D redraws.
  return canvas;
