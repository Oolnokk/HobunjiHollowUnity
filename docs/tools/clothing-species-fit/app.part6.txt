      'js/procedural-hand-outline-parity.js',
      'js/procedural-hand-scale-free-world.js',
      'js/procedural-hand-shoulder-aim.js',
      'config/character-rig-scale-defaults.js',
      'js/character-rig-scale.js',
      'js/character-rig-scale-avatar-runtime.js',
      'js/procedural-hand-frame-driver.js',
    ];

    for(let i=0;i<coreScripts.length;i++){
      const path=coreScripts[i];
      await executeRuntimeScript(path,i+1,coreScripts.length);
      if(path==='config/scratchbones-config.js'){
        const portrait=window.SCRATCHBONES_CONFIG?.game?.assets?.portrait;
        if(portrait){
          portrait.assetBase=new URL('assets/',docs).href;
          portrait.configBase=new URL('config/',docs).href;
        }
      }
      if(path==='js/portrait-utils.js')installCanonicalCloudMaskScale();
    }

    const required=[
      ['NpcAvatarPreview',window.NpcAvatarPreview],
      ['PNGPlaneAvatar',window.PNGPlaneAvatar],
      ['ProceduralLegAnimation',window.ProceduralLegAnimation],
      ['ProceduralHandAttachments',window.ProceduralHandAttachments],
      ['ProceduralHandFrameDriver',window.ProceduralHandFrameDriver],
    ];
    const missing=required.filter(([,value])=>!value).map(([name])=>name);
    if(missing.length)throw new Error('Runtime rig dependencies missing after load: '+missing.join(', '));

    E.threeStatus.textContent='Runtime modules loaded; fetching portrait configuration…';
    const cosmeticsPromise=window.NpcAvatarPreview.ensurePortraitCosmetics({
      assetBase:new URL('assets/',docs).href,
      configBase:new URL('config/',docs).href,
    });
    await Promise.race([
      cosmeticsPromise,
      new Promise((_,reject)=>setTimeout(()=>reject(new Error('Timed out loading portrait configuration')),20000))
    ]);

    window.applyHobunjiAttachmentRigProfileCorrections?.();
    E.threeStatus.textContent='Hobunji runtime rig loaded.';
    return true;
  })().catch(error=>{
    R.three.runtimeReadyPromise=null;
    E.threeStatus.textContent='Runtime load failed: '+(error?.message||error)+' — tap Retry runtime.';
    throw error;
  });
  return R.three.runtimeReadyPromise;
}
function resetRuntimeLoad(){
  // Modules already executed safely stay installed; retry simply resumes/re-executes
  // idempotent runtime modules against the same repository root.
  R.three.runtimeReadyPromise=null;
  R.three.portraitCache.clear();
  R.three.buildGeneration++;
  E.threeStatus.textContent='Retrying runtime…';
  scheduleThreePreview(0);
}

async function runtimePortraitSet(){
  await ensureRigRuntime();
  const profileKey=runtimeProfileKey();
  const mask=currentMaskTransform();
  const cacheKey=`${profileKey}::mask:${Number(mask.scaleX).toFixed(4)}:${Number(mask.scaleY).toFixed(4)}:${Number(mask.translateY).toFixed(4)}`;
  if(R.three.portraitCache.has(cacheKey))return R.three.portraitCache.get(cacheKey);

  const promise=(async()=>{
    const npc=blankRuntimeNpc();
    let profile=window.NpcAvatarPreview.buildProfileFromNpcExport(npc);
    if(!profile){
      // Keep the random/profile seed independent of mask sliders so changing the
      // mask never changes body colours or any other appearance choice.
      profile=window.NpcAvatarPreview.randomProfile(`clothing-fit:${profileKey}`,{
        speciesId:E.species.value,gender:E.gender.value
      });
    }
    if(!profile)throw new Error(`Could not build runtime portrait profile for ${E.species.value} ${E.gender.value}.`);

    for(const prop of ['hair','hairFront','hairBack','hairSide','hairSideL','eyes','upperFace','facialHair','hat','hood','pauldron','torsoCosmetic','armCosmetic']){
      if(profile[prop])profile[prop]={id:'none',label:'None',tintSlot:null,layers:[]};
    }

    const size=Number(window.SCRATCHBONES_CONFIG?.game?.assets?.pngPlaneAvatar?.previewPortraitCanvasSize)||200;
    const make=()=>Object.assign(document.createElement('canvas'),{width:size,height:size});
    const front=make(),back=make(),head=make();

    // This is the important V11 change: the preview transform is installed
    // while portrait-utils renders the WHOLE front/back portrait. Its canonical
    // destination-out pass therefore cuts base torso, base arms, torso clothing,
    // overwear, etc. exactly at the normal full-portrait mask stage.
    await withPreviewCloudMaskTransform(()=>window.NpcAvatarPreview.renderProfileToCanvas(
      front,profile,{seatId:'clothing-fit-runtime',forceEyesOpen:true}
    ));
    await withPreviewCloudMaskTransform(()=>window.NpcAvatarPreview.renderProfileToCanvas(
      back,profile,{seatId:'clothing-fit-runtime',portraitView:'behind',forceEyesOpen:true}
    ));

    // Keep the neck/head-only texture on the ordinary runtime path. The cloud
    // mask belongs to the body portrait and should not be authored by these sliders.
    await window.NpcAvatarPreview.renderProfileToCanvas(
      head,profile,{seatId:'clothing-fit-runtime',onlyHeadSprite:true,forceEyesOpen:true}
    );
    return{npc,profile,front,back,head,size};
  })();
  R.three.portraitCache.set(cacheKey,promise);
  try{return await promise}catch(error){R.three.portraitCache.delete(cacheKey);throw error}
}
function normalizedLayerXform(raw={}){
  return{
    ax:Number(raw.ax)||0,ay:Number(raw.ay)||0,
    sx:Number(raw.sx??raw.scaleX??raw.scaleMulX)||1,
    sy:Number(raw.sy??raw.scaleY??raw.scaleMulY)||1,
  };
}
function targetBodyLayerRecord(descriptor=null){
  const cfg=R.speciesCfg.get(E.species.value)?.[E.gender.value];
  const layers=cfg?.portraitBodyLayers||[];
  const wanted=String(descriptor?.layerId||descriptor?.role||'').toLowerCase();
  if(wanted){
    const exact=layers.find(layer=>String(layer?.id||'').toLowerCase()===wanted);
    if(exact)return exact;
    const normalizedWanted=wanted.replace(/[^a-z0-9]/g,'');
    const normalized=layers.find(layer=>String(layer?.id||'').toLowerCase().replace(/[^a-z0-9]/g,'')===normalizedWanted);
    if(normalized)return normalized;
  }
  return layers.find(layer=>String(layer.id||'').toLowerCase()==='torso')||layers[0]||null;
}
function targetBodyLayerXform(){
  const torso=targetBodyLayerRecord({layerId:'torso'});
  return normalizedLayerXform(torso?.xform||{ax:-.2,ay:0,scaleX:2.55,scaleY:2.55});
}
function routedBodyTintSlot(descriptor){
  const layer=targetBodyLayerRecord(descriptor);
  return String(layer?.tintSlot||descriptor?.tintSlot||'A')||'A';
}
function normalizedPortraitAssetKey(value){
  return String(value||'').replace(/[?#].*$/,'').replace(/^https?:\/\/[^/]+\//i,'').replace(/^\.?\/+/,'').replace(/^docs\//,'').replace(/^assets\//,''); // Used to match the selected body raster even when one side is absolute and the renderer uses a repository-relative source key.
}
function samePortraitAsset(a,b){
  const left=normalizedPortraitAssetKey(a),right=normalizedPortraitAssetKey(b);
  return !!(left&&right&&(left===right||left.endsWith('/'+right)||right.endsWith('/'+left)));
}
