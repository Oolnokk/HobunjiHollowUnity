  R.three.zoom=distance;
  R.three.camera.position.set(center.x,Math.max(center.y,size.y*.42),center.z+distance);
  R.three.camera.lookAt(center.x,Math.max(center.y,size.y*.42),center.z);
  R.three.camera.updateMatrixWorld(true);
}
function makeExtendedRigPlanes(THREE,canvas,modelWidth,modelHeight){
  if(!canvas)return null;
  const group=new THREE.Group();group.name='ImportedExtendedPortraitOverlay';
  const worldHeight=modelHeight*(canvas.height/200);
  const topY=modelHeight;
  const centerY=topY-worldHeight/2;
  const makeTexture=(source,name)=>{
    const texture=new THREE.CanvasTexture(source);texture.name=name;texture.generateMipmaps=false;
    texture.minFilter=THREE.LinearFilter;texture.magFilter=THREE.LinearFilter;texture.needsUpdate=true;return texture;
  };
  const frontTexture=makeTexture(canvas,'extended_overlay_front');
  const backTexture=makeTexture(flipCanvasX(canvas),'extended_overlay_back');
  const geometry=new THREE.PlaneGeometry(modelWidth,worldHeight);
  const front=new THREE.Mesh(geometry,new THREE.MeshBasicMaterial({map:frontTexture,transparent:true,alphaTest:.001,side:THREE.FrontSide,depthTest:true,depthWrite:false}));
  front.position.set(0,centerY,.008);front.name='extended_overlay_front_plane';group.add(front);
  const back=new THREE.Mesh(geometry.clone(),new THREE.MeshBasicMaterial({map:backTexture,transparent:true,alphaTest:.001,side:THREE.FrontSide,depthTest:true,depthWrite:false}));
  back.rotation.y=Math.PI;back.position.set(0,centerY,-.008);back.name='extended_overlay_back_plane';group.add(back);
  return group;
}
async function updateThreePreview(){
  if(!R.body||!ensureThreePreview())return;
  const buildId=++R.three.buildGeneration;
  try{
    await ensureRigRuntime();
    if(buildId!==R.three.buildGeneration)return;
    E.threeStatus.textContent='Rendering runtime portrait + cloud mask…';

    const avatar=await composeRuntimePortrait();
    if(buildId!==R.three.buildGeneration)return;
    const THREE=window.THREE;
    disposeRuntimeRig();

    const rigGroup=new THREE.Group();
    rigGroup.name=`RuntimeCharacterRig_${E.species.value}_${E.gender.value}`;
    const avatarCfg=window.SCRATCHBONES_CONFIG?.game?.assets?.pngPlaneAvatar||{};
    const baseWidth=Number(avatarCfg.worldModelWidth)>0?Number(avatarCfg.worldModelWidth):.9;
    const model=window.PNGPlaneAvatar.buildSinglePlaneAvatarModel(THREE,avatar.frontComposed,{
      backCanvas:avatar.backComposed,
      headCanvas:avatar.head,
      neckRig:true,
      profile:avatar.profile,
      appearance:avatar.npc.appearance,
      npcRecord:avatar.npc,
      modelWidth:baseWidth,
      modelHeight:baseWidth,
      name:'ClothingFitRuntimePortrait',
      userData:{source:'clothing-fit-runtime-preview',nonInteractive:true},
    });
    model.userData.proceduralHandParent=rigGroup;
    model.userData.rigAvatarProfile=avatar.profile;
    rigGroup.add(model);

    const modelHeight=Number(model.userData.portraitModelHeight)||baseWidth;
    const modelWidth=Number(model.userData.portraitModelWidth)||baseWidth;
    model.position.y=modelHeight/2;

    const feet=window.ProceduralLegAnimation?.attach?.(THREE,rigGroup,{
      speciesId:E.species.value,
      gender:E.gender.value,
      bodyColors:avatar.profile?.bodyColors||avatar.npc.appearance?.bodyColors,
      modelWidth,modelHeight,
      handAttachY:model.userData.handAttachY,
      name:'ClothingFitRuntimeFeet',
      profile:avatar.profile,
      portraitSize:avatar.frontComposed.width||200,
    })||null;
    feet?.update?.(0,0,false,null);

    if(E.showExtended3D.checked&&R.extendedSource){
      const extCanvas=await extendedRuntimeCanvas();
      if(buildId!==R.three.buildGeneration){feet?.dispose?.();disposeObjectTree(rigGroup);return}
      if(extCanvas){
        const extGroup=makeExtendedRigPlanes(THREE,extCanvas,modelWidth,modelHeight);
        if(extGroup){rigGroup.add(extGroup);R.three.extendedGroup=extGroup}
      }
    }

    R.three.group.add(rigGroup);
    R.three.rigRoot=rigGroup;R.three.avatarModel=model;R.three.feet=feet;

    // Match the full-character preview contract used elsewhere in the repository.
    const scale=window.HobunjiCharacterRigScale?.scaleFor?.(E.species.value,E.gender.value)||null;
    window.HobunjiCharacterRigScale?.applyToParent?.(rigGroup,E.species.value,E.gender.value,scale,0);
    window.ProceduralHandFrameDriver?.syncNow?.();

    rigGroup.updateMatrixWorld(true);
    R.three.group.rotation.set(R.three.rotX,R.three.rotY,0);
    frameRuntimeRig();
    const cloud=R.speciesCfg.get(E.species.value)?.[E.gender.value]?.portraitOpacityMaskLayer?.url?'cloud mask active':'no species cloud mask';
    const handState=model.userData?.proceduralHandRig?'hands attached':'hands queued';
    E.threeStatus.textContent=`Runtime rig: portrait + neck + ${handState} + feet + ${cloud}${R.extendedGroup?' + extended overlay':''}.`;
  }catch(error){
    if(buildId!==R.three.buildGeneration)return;
    console.error(error);E.threeStatus.textContent='Runtime rig preview failed: '+error.message;log('3D RUNTIME ERROR '+error.message);
  }
}
function scheduleThreePreview(delay=90){
  clearTimeout(R.three.buildTimer);
  R.three.buildTimer=setTimeout(()=>updateThreePreview(),delay);
}
function render(){
  if(!R.body)return;
  sizeStage();C.setTransform(1,0,0,1,0,0);C.clearRect(0,0,E.stage.width,E.stage.height);C.imageSmoothingEnabled=false;
  if(E.showBody.checked)C.drawImage(R.body,0,0);

  if(mode==='transform'||mode==='fit'||mode==='preview'){
    R.warped=currentWarp();
    const showRoutedSource=routedPatchIsBody()?E.showBody.checked:E.showGarment.checked;
    if(R.warped&&showRoutedSource){
      C.save();C.globalAlpha=routedPatchIsBody()?1:+E.opacity.value;C.drawImage(R.warped,0,0);C.restore();
    }
  }

  if(E.showAuthored.checked&&R.authoredTargetCloth){
    C.save();
    C.globalAlpha=+E.authoredOpacity.value;
    C.drawImage(R.authoredTargetCloth,0,0); // Existing hand-authored target-species art overlays the experimental warp for visual matching.
    C.restore();
  }

  if(E.showMeshes.checked){
    drawGrid(C,SOURCE_GRID,'#ffb84a',false); // Fixed Mao-ao source.
    drawGrid(C,transformedSourceGrid(),'#65d48a',mode==='transform'); // Recorded scale + translation.
    if(mode!=='transform')drawGrid(C,targetGrid(),'#45c3ff',mode!=='preview'); // Base transform plus local fit.
  }

  E.label.textContent=
    mode==='transform'
      ?`Base Transform • ${garmentRecord()?.label||''} • ${E.species.options[E.species.selectedIndex]?.text||E.species.value} ${E.gender.value}`
      :mode==='body'
        ?`Grid only • ${E.species.options[E.species.selectedIndex]?.text||E.species.value} ${E.gender.value}`
        :mode==='fit'
          ?`Species Fit • ${garmentRecord()?.label||''} • Mao-ao ${E.gender.value} → ${E.species.options[E.species.selectedIndex]?.text||E.species.value}`
          :`Preview • recorded base transform + local 6×6 deformation applied to ${garmentRecord()?.label||''}`;

  refreshHelp();refreshDebug();scheduleThreePreview();
}
