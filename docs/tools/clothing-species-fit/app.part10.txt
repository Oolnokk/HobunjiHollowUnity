function pointer(e){
  const r=E.stage.getBoundingClientRect();
  return{x:(e.clientX-r.left)*E.stage.width/r.width,y:(e.clientY-r.top)*E.stage.height/r.height};
}
function nearest(points,p){
  const px=toDisplayPixels(points,E.stage.width,E.stage.height);let best=-1,bd=Infinity;
  px.forEach((q,i)=>{const d=Math.hypot(q.x-p.x,q.y-p.y);if(d<bd){bd=d;best=i}});
  return bd<=HIT_RADIUS?best:-1;
}
function editable(){return mode==='fit'||mode==='body'}
let transformDrag=null;
E.stage.addEventListener('pointerdown',e=>{
  if(mode==='transform'){
    const p=pointer(e),tr=currentTransform();
    pushUndo();transformDrag={x:p.x,y:p.y,translateX:tr.translateX,translateY:tr.translateY};
    E.stage.setPointerCapture(e.pointerId);e.preventDefault();return;
  }
  if(!editable())return;
  const index=nearest(targetGrid(),pointer(e));if(index<0)return;
  dragIndex=index;dragSnapshotTaken=false;E.stage.setPointerCapture(e.pointerId);e.preventDefault();
});
E.stage.addEventListener('pointermove',e=>{
  if(mode==='transform'&&transformDrag){
    const p=pointer(e),tr=currentTransform();
    tr.translateX=transformDrag.translateX+(p.x-transformDrag.x);
    tr.translateY=transformDrag.translateY+(p.y-transformDrag.y);
    saveState();syncTransformControls();render();e.preventDefault();return;
  }
  if(dragIndex==null||!editable())return;
  if(!dragSnapshotTaken){pushUndo();dragSnapshotTaken=true}
  const p=pointer(e),baseGrid=transformedSourceGrid(),offsets=currentOffsets();
  const desired={
    x:Math.max(-0.5,Math.min(1.5,p.x/(E.stage.width-1))),
    y:Math.max(-0.5,Math.min(1.5,p.y/(E.stage.height-1)))
  };
  offsets[dragIndex]={x:desired.x-baseGrid[dragIndex].x,y:desired.y-baseGrid[dragIndex].y};
  saveState();render();e.preventDefault();
});
function endDrag(e){
  dragIndex=null;dragSnapshotTaken=false;transformDrag=null;
  if(e?.pointerId!=null&&E.stage.hasPointerCapture?.(e.pointerId))E.stage.releasePointerCapture(e.pointerId);
}
E.stage.addEventListener('pointerup',endDrag);E.stage.addEventListener('pointercancel',endDrag);
