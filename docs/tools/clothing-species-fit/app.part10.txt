function nearest(points,p){
  const px=toDisplayPixels(points,E.stage.width,E.stage.height);let best=-1,bd=Infinity;
  px.forEach((q,i)=>{const d=Math.hypot(q.x-p.x,q.y-p.y);if(d<bd){bd=d;best=i}});
  return bd<=HIT_RADIUS?best:-1;
}

function paintRearPoint(point){
  const canvas=R.headbandRearCanvas;if(!canvas)return;
  const ctx=canvas.getContext('2d',{alpha:true}),radius=Math.max(1,Math.round(+E.headbandRearBrush.value||1));
  const x=Math.round(point.x),y=Math.round(point.y);
  if(rearBrushAction==='sample'){
    if(x>=0&&x<canvas.width&&y>=0&&y<canvas.height){
      const rgba=ctx.getImageData(x,y,1,1).data;
      if(rgba[3])E.headbandRearColor.value='#'+[rgba[0],rgba[1],rgba[2]].map(n=>n.toString(16).padStart(2,'0')).join('');
    }
    rearBrushAction='draw';return;
  }
  ctx.fillStyle=E.headbandRearColor.value; // Author PNG pixels at full opacity, without raster interpolation.
  for(let dy=-radius;dy<=radius;dy++)for(let dx=-radius;dx<=radius;dx++){
    if(dx*dx+dy*dy>radius*radius)continue;
    const px=x+dx,py=y+dy;
    if(px<0||py<0||px>=canvas.width||py>=canvas.height)continue;
    if(rearBrushAction==='erase')ctx.clearRect(px,py,1,1);
    else ctx.fillRect(px,py,1,1);
  }
}

      if(!selected||selected.curve!==headbandDrag.curve)return;
      const desired={...headbandDrag.settings,
        x:headbandDrag.settings.x+p.x-headbandDrag.start.x,
        y:headbandDrag.settings.y+p.y-headbandDrag.start.y};
      const result=window.ClothingHeadbandSpline.fitWholeTransform(
        selected.entry.base,selected.entry.settings,desired,R.body.width,R.body.height);
      record[headbandDrag.curve]=result.nodes;selected.entry.settings=result.settings;
      if(result.fraction>0)record.enabled=true; // Whole-curve drag previews the fitted headband, even if its spline was previously off.
      R.headbandWholeWarning=result.fraction<.99999?'Whole-curve drag constrained at sprite edge.':'';
      syncHeadbandWholeControls();
    }else{
      record[headbandDrag.curve]=window.ClothingHeadbandSpline.dragNode(
        record[headbandDrag.curve],headbandDrag.index,p.x/Math.max(1,E.stage.width-1),
        p.y/Math.max(1,E.stage.height-1),E.stage.width,E.stage.height
      ); // Individual nodes cannot cross neighbors.
    }
    queueHeadbandRender();e.preventDefault();return;
  }
  if(mode==='protect'&&contourStroke){
    contourPaintAt(pointer(e));e.preventDefault();return;
  }
  if(mode==='quad'){if(quadPointerMove(e))e.preventDefault();return}
  if(mode==='hole'){if(holePointerMove(e))e.preventDefault();return}
  if(mode==='pin'){if(pinPointerMove(e))e.preventDefault();return}
  if(mode==='transform'&&transformDrag){
    const p=pointer(e),tr=currentTransform();
    tr.translateX=transformDrag.translateX+(p.x-transformDrag.x);
    tr.translateY=transformDrag.translateY+(p.y-transformDrag.y);
    saveState();syncTransformControls();render();e.preventDefault();return;
  }
  if(dragIndex==null||!editable())return;
  if(!dragSnapshotTaken){pushUndo();dragSnapshotTaken=true}
  const p=pointer(e),baseGrid=transformedSourceGrid(),offsets=currentOffsets();
  const desired={
    x:Math.max(-0.5,Math.min(1.5,p.x/(E.stage.width-1))),
    y:Math.max(-0.5,Math.min(1.5,p.y/(E.stage.height-1)))
  };
  offsets[dragIndex]={x:desired.x-baseGrid[dragIndex].x,y:desired.y-baseGrid[dragIndex].y};
  saveState();render();e.preventDefault();
});
function endDrag(e){
  if(rearStroke){
    rearStroke=null;saveHeadbandRearArt();render();
    log('Committed rear sprite pixels to saved PNG for '+headbandSelectionKey()+'.');
  }
  if(headbandDrag){
    const record=currentHeadbandViewRecord();
    if(record&&!headbandDrag.whole)bakeHeadbandWhole(record,headbandDrag.curve); // Individual node edits become the new neutral whole-curve baseline.
    const movedWhole=headbandDrag.whole;
    headbandDrag=null;saveState();render();syncHeadbandControls();
    log('Saved '+(movedWhole?'whole-spline translation':'seven-point node edit')+' for '+headbandSelectionKey()+'.');
  }
  if(contourStroke){
    saveContourMask();syncContourControls();render(); // 3D rebuild is deferred until the brush is released, not once per touch event.
    contourStroke=null;
    log('Protected pixels saved for '+contourSelectionKey()+'.');
  }
  quadPointerUp();holePointerUp();pinPointerUp();
  dragIndex=null;dragSnapshotTaken=false;transformDrag=null;
  if(e?.pointerId!=null&&E.stage.hasPointerCapture?.(e.pointerId))E.stage.releasePointerCapture(e.pointerId);
}

function setTransformField(field,value,control){
  if(String(value).trim()==='')return false; // Editing an empty mobile number field must not collapse the sprite to zero width.
  const n=Number(value);
  if(!Number.isFinite(n))return false;
  const min=String(control?.min??'').trim(),max=String(control?.max??'').trim(); // Follow the actual paired control bounds.
  if((min&&n<Number(min))||(max&&n>Number(max)))return false;
  if((field==='scaleX'||field==='scaleY')&&n<=0)return false;
  currentTransform()[field]=n;saveState();syncTransformControls();render();return true;
}
