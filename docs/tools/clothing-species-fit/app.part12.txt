// ---------------------------------------------------------------------------
// Simple quad deformation: four corner handles. Dragging a corner spreads its
// movement over all 36 nodes with that corner's bilinear weight, so it layers
// on top of the base transform and any existing local node edits.
// ---------------------------------------------------------------------------
const QUAD_CORNERS=[0,GRID_SIZE-1,GRID_SIZE*(GRID_SIZE-1),GRID_SIZE*GRID_SIZE-1]; // Top-left, top-right, bottom-left, bottom-right grid indices.
const FULL_QUAD_BOX={u0:0,v0:0,u1:1,v1:1};
// The quad is a plain square on the item's bounding box (source-normalized);
// nodes beyond the box follow its nearest edge.
function quadCornerWeight(index,corner,box=FULL_QUAD_BOX){
  const p=SOURCE_GRID[index];
  const u=Math.max(0,Math.min(1,(p.x-box.u0)/Math.max(1e-6,box.u1-box.u0)));
  const v=Math.max(0,Math.min(1,(p.y-box.v0)/Math.max(1e-6,box.v1-box.v0)));
  return [(1-u)*(1-v),u*(1-v),(1-u)*v,u*v][corner];
}
function applyQuadCornerDelta(offsets,corner,dx,dy,box=FULL_QUAD_BOX){
  return offsets.map((o,i)=>{const w=quadCornerWeight(i,corner,box);return{x:o.x+w*dx,y:o.y+w*dy}});
}
let quadDrag=null;
function quadSourceBox(){
  const cloth=R.cloth; // Whole item (back + front layers), so the square encloses everything that moves.
  if(!cloth)return FULL_QUAD_BOX;
  if(R.quadBoxCache?.cloth===cloth)return R.quadBoxCache.box;
  const w=cloth.width,h=cloth.height,data=cloth.getContext('2d').getImageData(0,0,w,h).data;
  let x0=w,y0=h,x1=-1,y1=-1;
  for(let y=0;y<h;y++)for(let x=0;x<w;x++)if(data[(y*w+x)*4+3]>40){if(x<x0)x0=x;if(x>x1)x1=x;if(y<y0)y0=y;if(y>y1)y1=y}
  const box=x1<0?FULL_QUAD_BOX:{u0:x0/(w-1),v0:y0/(h-1),u1:x1/(w-1),v1:y1/(h-1)};
  R.quadBoxCache={cloth,box};
  return box;
}
function mapSourcePoint(grid,u,v){
  // Bilinear lookup of a source-normalized point through the fitted 6×6 grid.
  const n=GRID_SIZE-1,fx=Math.max(0,Math.min(n-1e-9,u*n)),fy=Math.max(0,Math.min(n-1e-9,v*n));
  const c=Math.floor(fx),r=Math.floor(fy),s=fx-c,t=fy-r;
  const p00=grid[r*GRID_SIZE+c],p10=grid[r*GRID_SIZE+c+1],p01=grid[(r+1)*GRID_SIZE+c],p11=grid[(r+1)*GRID_SIZE+c+1];
  return{x:(1-s)*(1-t)*p00.x+s*(1-t)*p10.x+(1-s)*t*p01.x+s*t*p11.x,y:(1-s)*(1-t)*p00.y+s*(1-t)*p10.y+(1-s)*t*p01.y+s*t*p11.y};
}
function quadDisplayCorners(){
  const box=quadSourceBox(),grid=targetGrid();
  return [[box.u0,box.v0],[box.u1,box.v0],[box.u0,box.v1],[box.u1,box.v1]].map(([u,v])=>{
    const p=mapSourcePoint(grid,u,v);return{x:p.x*(E.stage.width-1),y:p.y*(E.stage.height-1)};
  });
}
function quadPointerDown(e){
  const p=pointer(e),corners=quadDisplayCorners();
  let corner=-1,best=Infinity;
  corners.forEach((q,c)=>{const d=Math.hypot(q.x-p.x,q.y-p.y);if(d<best){best=d;corner=c}});
  if(corner<0||best>HIT_RADIUS*2)return false;
  pushUndo();
  quadDrag={corner,start:p,box:quadSourceBox(),offsets:currentOffsets().map(o=>({x:o.x,y:o.y}))};
  E.stage.setPointerCapture(e.pointerId);
  return true;
}
function quadPointerMove(e){
  if(!quadDrag)return false;
  const p=pointer(e);
  const dx=(p.x-quadDrag.start.x)/Math.max(1,E.stage.width-1),dy=(p.y-quadDrag.start.y)/Math.max(1,E.stage.height-1);
  A.localOffsets[ensureTargetData(E.species.value,E.gender.value)]=applyQuadCornerDelta(quadDrag.offsets,quadDrag.corner,dx,dy,quadDrag.box);
  saveState();render();
  return true;
}
function quadPointerUp(){
  if(!quadDrag)return;
  quadDrag=null;saveState();log('Saved quad corner deformation for '+targetKey(E.species.value,E.gender.value)+'.');
}
function drawQuadHandles(ctx){
  const q=quadDisplayCorners();
  ctx.save();ctx.strokeStyle='#ffe066';ctx.fillStyle='#ffe066';ctx.lineWidth=1.4;
  ctx.beginPath();[0,1,3,2,0].forEach((c,i)=>{i?ctx.lineTo(q[c].x,q[c].y):ctx.moveTo(q[c].x,q[c].y)});ctx.stroke();
  const side=HANDLE_RADIUS*2;
  for(const p of q)ctx.fillRect(p.x-side/2,p.y-side/2,side,side); // Square handles.
  ctx.restore();
}

// Fit recipes: a reusable base transform + 6×6 local offsets that can be
// applied to the current item or to several items at once. Recipes come from
// the repository library (config/clothing-fit-recipes/index.json) or are
// saved in this browser, and import/export as JSON.
// ---------------------------------------------------------------------------
const RECIPE_FORMAT='hobunji-clothing-fit-recipe-v1';
function validRecipe(recipe){
  const tr=recipe?.transform;
  return !!recipe&&recipe.format===RECIPE_FORMAT&&typeof recipe.name==='string'&&!!tr&&
    ['scaleX','scaleY','translateX','translateY'].every(key=>Number.isFinite(+tr[key]))&&+tr.scaleX>0&&+tr.scaleY>0&&
    Array.isArray(recipe.offsets)&&recipe.offsets.length===SOURCE_GRID.length&&
    recipe.offsets.every(point=>Number.isFinite(+point?.x)&&Number.isFinite(+point?.y));
}
function recipeFromCurrentFit(name,description=''){
  const fit=captureFitState();
  return{
    format:RECIPE_FORMAT,id:safeName(name).toLowerCase()||'recipe-'+Date.now(),name,description,region:fitRegion(),
    source:{species:E.sourceSpecies.value,gender:E.sourceGender.value},
    target:{species:E.species.value,gender:E.gender.value},
    transform:fit.transform,offsets:fit.offsets,
    ...(fit.hole&&fit.hole.enabled?{hole:{rate:fit.hole.rate,corners:fit.hole.corners}}:{}), // Optional hood-opening quad, in sprite pixels relative to each item's detected opening.
    createdAt:new Date().toISOString()
  };
}
function applyRecipeToCurrentFit(recipe){
  if(!validRecipe(recipe))throw new Error('Not a valid fit recipe.');
  const key=ensureTargetData(E.species.value,E.gender.value);
  const tr=recipe.transform;
  A.baseTransforms[key]={scaleX:+tr.scaleX,scaleY:+tr.scaleY,translateX:+tr.translateX,translateY:+tr.translateY};
  A.localOffsets[key]=recipe.offsets.map(point=>({x:+point.x,y:+point.y}));
  if(recipe.hole&&Array.isArray(recipe.hole.corners)){
    A.holeDeforms ||= {};
    A.holeDeforms[key]={enabled:true,rate:Number.isFinite(+recipe.hole.rate)?+recipe.hole.rate:1.5,
      corners:[0,1,2,3].map(i=>({x:+recipe.hole.corners[i]?.x||0,y:+recipe.hole.corners[i]?.y||0}))};
  }
  return key;
}

function drawHoleLayerView(C){

  const drawLayer=layer=>{

    t.globalCompositeOperation='color';t.fillStyle=color;t.fillRect(0,0,W,H); // Hue from the layer colour, light and dark (ink) from the art.

function drawHoleOverlay(ctx){
  const profile=currentHoleProfile();
  if(!profile){syncHoleStatus(null);return}
  const hole=currentHoleRecord(),Hole=window.ClothingHoodHoleDeform;
  const corners=Hole.normalizeCorners(hole.corners),rate=Hole.normalizeRate(hole.rate);
  const ring=(bins,displaced)=>bins.map((r,b)=>{
    const a=(b/bins.length)*2*Math.PI-Math.PI,x=profile.seed.x+Math.cos(a)*r,y=profile.seed.y+Math.sin(a)*r;
    if(!displaced)return{x,y};
    const v=Hole.displacementAt(profile,corners,rate,x,y);
    return{x:x+v.x,y:y+v.y};
  });
  const trace=points=>points.forEach((q,i)=>i?ctx.lineTo(q.x,q.y):ctx.moveTo(q.x,q.y));
  const outline=ring(profile.outline),rim=ring(profile.rim);
  const active=!!(hole.enabled&&!Hole.isIdentity(hole.corners));
  ctx.save();ctx.lineJoin='round';ctx.lineCap='round';
  // Falloff band: the cloth that moves with the opening, as a soft tint.
  ctx.beginPath();trace(outline);ctx.closePath();trace(rim);ctx.closePath();
  ctx.fillStyle='rgba(244,114,182,.10)';ctx.fill('evenodd');
  ctx.beginPath();trace(outline);ctx.closePath();
  ctx.setLineDash([3,2.2]);ctx.strokeStyle='rgba(244,114,182,.85)';ctx.lineWidth=.8;ctx.stroke();ctx.setLineDash([]);
  ctx.beginPath();trace(rim);ctx.closePath();
  ctx.strokeStyle=active?'rgba(34,211,238,.45)':'rgba(34,211,238,.95)';ctx.lineWidth=active?.8:1.1;ctx.stroke();
  if(active){ // Where the opening ends up, over a dark halo so it reads on any art.
    ctx.beginPath();trace(ring(profile.rim,true));ctx.closePath();
    ctx.strokeStyle='rgba(0,0,0,.55)';ctx.lineWidth=2.6;ctx.stroke();
    ctx.strokeStyle='#67e8f9';ctx.lineWidth=1.2;ctx.stroke();
  }
  const handles=holeHandlePoints(profile,hole);
  ctx.beginPath();[0,1,3,2].forEach((c,i)=>i?ctx.lineTo(handles[c].x,handles[c].y):ctx.moveTo(handles[c].x,handles[c].y));ctx.closePath();
  ctx.fillStyle='rgba(255,224,102,.07)';ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,.5)';ctx.lineWidth=2.2;ctx.stroke();
  ctx.strokeStyle='rgba(255,224,102,.9)';ctx.lineWidth=1;ctx.stroke();
  handles.forEach((q,i)=>{
    const r=HANDLE_RADIUS*(holeDrag?.corner===i?.95:.72);
    ctx.beginPath();ctx.arc(q.x,q.y,r,0,Math.PI*2);
    ctx.fillStyle='#ffe066';ctx.fill();ctx.strokeStyle='rgba(0,0,0,.65)';ctx.lineWidth=1.2;ctx.stroke();
    ctx.beginPath();ctx.arc(q.x-r*.28,q.y-r*.28,r*.32,0,Math.PI*2);ctx.fillStyle='rgba(255,255,255,.75)';ctx.fill();
  });
  ctx.restore();
  syncHoleStatus(profile);
}

function paintPinsAt(p,rear=false){
  const rec=currentPinRecord(),list=pinList(rec,rear),radius=pinBrushRadius();
  if(E.pinErase?.checked){const keep=list.filter(q=>Math.hypot(q.x-p.x,q.y-p.y)>radius);list.length=0;list.push(...keep);return}
  const mask=pinSnapMask(rear),Pins=window.ClothingSilhouettePins;if(!mask||!Pins)return;

  for(const q of Pins.outlinePixelsNear(mask.data,mask.width,mask.height,p.x,p.y,radius)){

  }

// Snap every front pin to the nearest rear-art outline pixel (same stored
// orientation, so the same side), within a few pixels.
function copyFrontPinsToRear(maxDistance=8){
  const rec=currentPinRecord(),mask=pinSnapMask(true),Pins=window.ClothingSilhouettePins;
  if(!mask||!Pins)return 0;
  const rear=pinList(rec,true),have=new Set(rear.map(q=>q.x+','+q.y));let added=0;
  for(const p of rec.points){
    let best=null,bestD=Infinity;
    for(const q of Pins.outlinePixelsNear(mask.data,mask.width,mask.height,p.x,p.y,maxDistance)){const d=Math.hypot(q.x-p.x,q.y-p.y);if(d<bestD){bestD=d;best=q}}
    if(best&&!have.has(best.x+','+best.y)){have.add(best.x+','+best.y);rear.push(best);added++}
  }
  return added;
}
