    for(const [index,leaf] of collectPatchImageNodes(sourceVariant).entries()){
      const targetNode=targetVariant?patchNodeAtPath(targetVariant,leaf.path):null;
      const labelPath=leaf.path.join(' › ')||`layer ${index+1}`;
      items.push({
        id:`clothing:${garment.id}:${gender}:${index}`,
        kind:'clothing',
        garmentId:garment.id,
        garmentLabel:garment.label,
        label:`${garment.label} • ${labelPath}`,
        layerPath:[...leaf.path],
        sourceSpecies:'mao-ao',
        targetSpecies,
        gender,
        sourceUrl:leaf.url,
        targetUrl:targetNode?.image?.url||null,
        role:leaf.role||'',
        view:inferPatchView(leaf.path,leaf.url)
      });
    }
  }

  const targetBody=R.speciesCfg.get(targetSpecies)?.[gender]?.portraitBodyLayers||[];
  const targetSpeciesLabel=E.species.options[E.species.selectedIndex]?.text||targetSpecies; // Used in the body-patch selector so species changes are visible immediately rather than continuing to read Mao-ao.
  for(const [index,layer] of targetBody.entries()){
    if(!layer?.url)continue;
    items.push({
      id:`body:${gender}:${layer.id||index}`,
      kind:'body',
      layerId:layer.id||`layer${index+1}`,
      label:`Body • ${layer.id||`layer ${index+1}`} • ${targetSpeciesLabel} ${gender}`,
      sourceSpecies:targetSpecies,
      targetSpecies,
      gender,
      sourceUrl:layer.url,
      targetUrl:layer.url,
      role:layer.id||'',
      tintSlot:layer?.tintSlot||'A',
      view:'front',
      nativeBodyPatch:true
    });
  }
  return items;
}
async function loadBehindPatchSource(id){
  const descriptor=behindPatchCatalog().find(item=>item.id===id);
  if(!descriptor)throw new Error('That behind-sprite source is no longer available. Refresh the source list.');
  const image=await loadImage(assetUrl(descriptor.sourceUrl));
  const canvas=document.createElement('canvas'); // Native source pixels are preserved before the current 6×6 deformation is applied.
  canvas.width=image.naturalWidth||image.width;canvas.height=image.naturalHeight||image.height;
  const x=canvas.getContext('2d',{alpha:true});x.imageSmoothingEnabled=false;x.drawImage(image,0,0);
  canvas._paths=[descriptor.sourceUrl];
  return{descriptor,canvas};
}
function warpBehindPatchCanvas(sourceCanvas,fitState=null){
  if(!sourceCanvas||!R.body)throw new Error('The clothing species-fit editor is not ready to deform a behind sprite yet.');
  return warpGarment(sourceCanvas,fitState?targetGridFromFitState(fitState):targetGrid(),R.body.width,R.body.height); // Export callers can pin the exact base transform + 6×6 fit active when the job was authored.
}
function setMainWorkflowPatch(sourceCanvas,meta={}){
  if(!sourceCanvas)throw new Error('Cannot route an empty patch source into the main workflow.');
  R.patchWorkflow={
    active:true,canvas:sourceCanvas,descriptor:meta.descriptor||null,
    filename:String(meta.filename||''),patchName:String(meta.patchName||''),revision:(R.patchWorkflow.revision||0)+1
  }; // Clothing receives a source+patch composite; body receives patch-only pixels. Both are routed through the same live transform/6×6 warp before avatar assembly.
  E.clothPng.textContent=meta?.descriptor?.kind==='body'?'Export warped body patch PNG':'Export patched clothing-back PNG';
  render();
}
function clearMainWorkflowPatch(){
  if(!R.patchWorkflow.active&&!R.patchWorkflow.canvas)return;
  R.patchWorkflow={active:false,canvas:null,descriptor:null,filename:'',patchName:'',revision:(R.patchWorkflow.revision||0)+1};
  E.clothPng.textContent='Export fitted clothing PNG';
  render();
}
