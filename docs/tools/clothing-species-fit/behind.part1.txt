const crcTable=(()=>{
  const table=new Uint32Array(256); // Shared CRC-32 lookup table for the dependency-free ZIP writer.
  for(let n=0;n<256;n++){
    let c=n; // Builds one standard ZIP CRC-32 table entry.
    for(let k=0;k<8;k++)c=(c&1)?(0xedb88320^(c>>>1)):(c>>>1);
    table[n]=c>>>0;
  }
  return table;
})();
