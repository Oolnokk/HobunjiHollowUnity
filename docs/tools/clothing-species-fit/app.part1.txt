function makeRegularGrid(size){
  const points=[]; // Creates row-major normalized points that touch all four portrait edges.
  for(let row=0;row<size;row++){
    for(let col=0;col<size;col++){
      points.push({x:col/(size-1),y:row/(size-1)});
    }
  }
  return points;
}

function migrateLegacyV1(){
  try{
    const old=JSON.parse(localStorage.getItem('hobunji_clothing_species_fit_6x6_v1')||'null');
    if(!old||old.version!==1||old.gridSize!==GRID_SIZE)return null;
    const next=fresh();
    for(const [key,grid] of Object.entries(old.targetGrids||{})){
      if(!Array.isArray(grid)||grid.length!==SOURCE_GRID.length)continue;
      next.baseTransforms[key]=defaultTransform();
      next.localOffsets[key]=grid.map((p,i)=>({x:p.x-SOURCE_GRID[i].x,y:p.y-SOURCE_GRID[i].y}));
    }
    next.settings={...next.settings,...(old.settings||{}),mode:'fit'};
    return next;
  }catch{return null}
}

    }
    return migrateLegacyV1()||fresh();
  }catch{return migrateLegacyV1()||fresh()}
}

function assetUrl(path){
  const p=String(path||'');
  if(/^https?:/i.test(p))return p;
  if(p.startsWith('./assets/'))return docsUrl(p.slice(2));
  if(p.startsWith('assets/'))return docsUrl(p);
  return docsUrl('assets/'+p.replace(/^\.\//,''));
}
