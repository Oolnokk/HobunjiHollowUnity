function archiveAssetPath(url){
  const path=String(url||'').replace(/^\.\//,'');
  if(!/^assets\/.+\.png$/i.test(path)||/[\\?#:"<>|\x00-\x1f]/.test(path)||
     path.split('/').some(segment=>!segment||segment==='.'||segment==='..'))return null;
  return path;
}
function fittedFallbackAssetPath(layer,selection){
  const source=archiveAssetPath(layer.url);
  const directory=source?.slice(0,source.lastIndexOf('/')+1)||'assets/cosmetics/clothes/';
  const original=(source?.split('/').pop()||safeName(layer.name)+'.png').replace(/\.png$/i,'');
  const noSourceGender=original.replace(/[_-](male|female|m|f)$/i,'');
  const suffix=selection.targetGender==='female'?'f':'m';
  return directory+noSourceGender+'_'+safeName(selection.targetSpecies)+'_'+suffix+'.png';
}
function archiveVariantLeaf(variant,path,originalLeaf){
  let cursor=variant,created=false;
  for(let index=0;index<path.length;index++){
    const key=path[index];
    if(!cursor[key]||typeof cursor[key]!=='object'){
      cursor[key]=index===path.length-1?JSON.parse(JSON.stringify(originalLeaf)):{};
      created=true;
    }
    cursor=cursor[key];
  }
  return{leaf:cursor,created};
}
function assetReferencedOutsideVariant(path,garmentId,targetVariantKey){
  const matches=node=>{
    if(!node||typeof node!=='object')return false;
    if(archiveAssetPath(node.image?.url)===path||archiveAssetPath(node.behindImage?.url)===path)return true;
    return Object.entries(node).some(([key,value])=>
      key!=='image'&&key!=='behindImage'&&value&&typeof value==='object'&&matches(value));
  };
  for(const garment of R.garments){
    for(const [variantKey,variant] of Object.entries(garment.cfg?.speciesVariants||{})){
      if(garment.id===garmentId&&variantKey===targetVariantKey)continue;
      if(matches(variant))return true;
    }
    if(garment.id!==garmentId&&matches(garment.cfg?.parts))return true;
  }
  return false; // Shared texture URLs must not overwrite an unrelated species or clothing item.
}

function assetReferencedByOtherTargetLayer(assetPath,variant,layerPath){
  // Every reference to the same PNG must belong to the layer being fitted.
  // image and behindImage on that SAME layer may intentionally share an identical PNG.
  const references=[];
  const visit=(node,path=[])=>{
    if(!node||typeof node!=='object')return;
    if(archiveAssetPath(node.image?.url)===assetPath)references.push(path);
    if(archiveAssetPath(node.behindImage?.url)===assetPath)references.push(path);
    for(const [key,value] of Object.entries(node)){
      if(key==='image'||key==='behindImage')continue;
      if(value&&typeof value==='object')visit(value,[...path,key]);
    }
  };
  visit(variant);
  const ownerPath=layerPath.at(-1)==='behindImage'?layerPath.slice(0,-1):layerPath;
  return references.some(path=>JSON.stringify(path)!==JSON.stringify(ownerPath));
} // Avoid clobbering unexported or differently authored layers within the target variant.

function archiveAssetUsedByAnyGarment(path){
  return R.garments.some(garment=>
    Object.values(garment.cfg?.speciesVariants||{}).some(variant=>
      assetReferencedByOtherTargetLayer(path,variant,[]))||
    assetReferencedByOtherTargetLayer(path,{parts:garment.cfg?.parts||{}},[])
  );
} // Even suggested new filenames cannot overwrite another already-authored sprite URL.

function buildFittedLayerArchivePlan(layers,selection){
  const garment=garmentRecord(),cfg=garment?.cfg;
  if(!cfg||!garment.configPath)throw new Error('This garment has no repository cosmetic config path for a ready-to-install export.');
  const sourceKey=selection.sourceSpecies+'_'+selection.sourceGender;
  const targetKey=selection.targetSpecies+'_'+selection.targetGender;
  const sourceVariant=cfg.speciesVariants?.[sourceKey]||{parts:cfg.parts||{}};
  if(!sourceVariant&&!HEAD_SLOTS.has(garment.slot))throw new Error('Cannot resolve the selected source garment variant.');
  const config=JSON.parse(JSON.stringify(cfg));
  config.speciesVariants ||= {};
  let modified=false;
  if(!config.speciesVariants[targetKey]){config.speciesVariants[targetKey]={parts:{}};modified=true}
  const draft=config.speciesVariants[targetKey];
  const used=new Set(),files=[];
  for(const layer of layers){
    const sourceLeaf=patchNodeAtPath(sourceVariant,layer.path)||
      (layer.path.at(-1)==='behindImage'?{url:layer.url}:{image:{url:layer.url},layerRole:layer.role||''});
    const originalTarget=patchNodeAtPath(cfg.speciesVariants?.[targetKey],layer.path);
    const behind=layer.path.at(-1)==='behindImage';
    const targetUrl=behind?originalTarget?.url:originalTarget?.image?.url;
    const existing=archiveAssetPath(targetUrl);
    const shared=existing&&(
      assetReferencedOutsideVariant(existing,garment.id,targetKey)||
      assetReferencedByOtherTargetLayer(existing,cfg.speciesVariants?.[targetKey],layer.path)||
      assetReferencedByOtherTargetLayer(existing,{parts:cfg.parts||{}},[])
    );
    let outputName=existing&&!shared?existing:fittedFallbackAssetPath(layer,selection);
    const collides=name=>used.has(name.toLowerCase())||
      (name!==existing&&archiveAssetUsedByAnyGarment(name));
    if(collides(outputName)){
      const suffix=safeName(layer.path.join('_'));
      const base=outputName.replace(/\.png$/i,'');
      outputName=base+'__'+suffix+'.png';
      for(let index=2;collides(outputName);index++)outputName=base+'__'+suffix+'_'+index+'.png';
    }
    used.add(outputName.toLowerCase());
    const changedUrl=!existing||outputName!==existing;
    const installed=archiveVariantLeaf(draft,layer.path,sourceLeaf);
    if(installed.created)modified=true;
    if(changedUrl){
      if(behind)installed.leaf.url='./'+outputName;
      else{
        installed.leaf.image ||= {};
        installed.leaf.image.url='./'+outputName;
      }
      modified=true;
    }
    // If front and rear originally use the same image, updating a front file must
    // update its matching rear alias even when the target layer ALREADY exists.
    // A distinct authored behindImage is handled by its own exported layer instead.
    if(!behind&&changedUrl&&installed.leaf.behindImage?.url===targetUrl){
      const explicitRear=layers.some(other=>other.path.length===layer.path.length+1&&
        other.path.at(-1)==='behindImage'&&
        JSON.stringify(other.path.slice(0,-1))===JSON.stringify(layer.path));
      if(!explicitRear)installed.leaf.behindImage.url='./'+outputName;
    }
    if(installed.created&&!behind&&installed.leaf.behindImage?.url===layer.url){
      installed.leaf.behindImage.url='./'+outputName;
      modified=true;
    }
    files.push({
      outputName,originalLayerName:layer.name,sourceConfigPath:layer.path,sourceUrl:layer.url,
      targetUrl:'./'+outputName,existingTargetFilename:!!existing&&!changedUrl,
      sharedTargetAssetAvoided:!!shared,
      layerRole:layer.role,view:layer.view,patched:layer.patched
    });
  }
  return{files,configPath:modified?garment.configPath:null,originalConfigPath:garment.configPath,
    config:modified?config:null};
}

async function exportFittedGarment(){
  const fit=captureFitState();
  if(!R.cloth)return;
  if((R.cloth._layers?.length||0)<=1||routedPatchIsBody()){
    const output=clothOut(fit); // Don't spend another 800-triangle warp on a single-layer PNG.
    if(output){downloadCanvas(output,fittedClothingFilename());log(`Exported single fitted PNG (${fit.key}).`)}
    return;
  }
  const layers=fittedLayerOutputs(fit);
  const zipTools=window.ClothingFitZip; // Reuses the behind-sprite author's existing dependency-free ZIP encoder.
  if(!zipTools){log('ERROR ZIP export helper is not loaded; the editor cannot export layered sprites safely.');return}
  const selection={garmentId:E.garment.value,fitRegion:fitRegion(),sourceSpecies:E.sourceSpecies.value,
    sourceGender:E.sourceGender.value,targetSpecies:E.species.value,targetGender:E.gender.value}; // Pinned before asynchronous mobile PNG encoding.
  const stem=`${safeName(selection.garmentId)}_${safeName(selection.targetSpecies)}_${selection.targetGender}`;
  const entries=[];
  E.clothPng.disabled=true;
  try{
    const plan=buildFittedLayerArchivePlan(layers,selection);
    for(let index=0;index<layers.length;index++){
      const mapped=plan.files[index];
      E.sourceLayerStatus.textContent='Baking '+(index+1)+'/'+layers.length+': '+mapped.outputName;
      entries.push({name:mapped.outputName,bytes:await zipTools.canvasToPngBytes(layers[index].canvas)});
    }
    if(plan.configPath)
      entries.push({name:plan.configPath,bytes:new TextEncoder().encode(JSON.stringify(plan.config,null,2)+'\n')});
    const manifest={
      format:'hobunji-clothing-fit-layered-v2',...selection,
      installRoot:'docs/',targetConfigUpdated:!!plan.configPath,
      configPath:plan.originalConfigPath, // Pin garment metadata before asynchronous image encoding.
      fit:{key:fit.key,baseTransform:fit.transform,localOffsets:fit.offsets},
      files:plan.files
    };
    entries.push({name:'fitted-layers-manifest.json',bytes:new TextEncoder().encode(JSON.stringify(manifest,null,2))});
    const readme=[
      'HOBUNJI HOLLOW — FITTED CLOTHING LAYERS',
      '',
      'Extract this ZIP into the repository docs/ directory, preserving its subfolders.',
      'PNG filenames and folders match the target cosmetic config image URLs.',
      plan.configPath
        ?'The included '+plan.configPath+' adds/updates target layer references. Merge the JSON with newer repository edits if the repository changed since export.'
        :'All image URLs already exist in the target cosmetic config; no config update is needed.',
      'Existing PNGs with the same names will be overwritten. Back them up first.',
      'See fitted-layers-manifest.json for every original layer name and source/target path.'
    ].join('\n');
    entries.push({name:'README-FITTED-LAYERS.txt',bytes:new TextEncoder().encode(readme)});
    const archiveName=`${stem}_fitted_layers.zip`;
    zipTools.downloadBlob(zipTools.buildStoredZip(entries),archiveName);
    log(`Exported ${plan.files.length} ready-to-install layers in ${archiveName} for ${selection.targetSpecies} ${selection.targetGender}.`);
    E.sourceLayerStatus.textContent=`Exported ${plan.files.length} layers with source-native names.${plan.configPath?' Updated cosmetic config included.':''}`;
  }catch(error){
    log('ERROR Fitted layer ZIP export: '+error.message);
    E.sourceLayerStatus.textContent='Layer export failed: '+error.message;
  }finally{E.clothPng.disabled=!R.cloth}
}
function overlayOut(){

  const output=document.createElement('canvas');output.width=R.body.width;output.height=R.body.height;
  const ctx=output.getContext('2d',{alpha:true});ctx.imageSmoothingEnabled=false;
  if(fitRegion()==='headwear'){
    const back=headwearWarpForView('under',true),front=headwearWarpForView('front',true);
    if(back)ctx.drawImage(back,0,0);
    ctx.drawImage(bodyOut(),0,0);
    if(front)ctx.drawImage(front,0,0);
  }else{ctx.drawImage(R.body,0,0);ctx.drawImage(clothing,0,0)}
  return output;
}
function safeName(value){return String(value||'sprite').trim().replace(/[^a-z0-9_-]+/gi,'_').replace(/^_+|_+$/g,'')||'sprite'}
function downloadCanvas(canvas,filename){
  if(!canvas)return;
  const save=url=>{const link=document.createElement('a');link.href=url;link.download=filename;link.click();setTimeout(()=>URL.revokeObjectURL(url),1500)};
  if(canvas.toBlob){canvas.toBlob(blob=>{if(blob)save(URL.createObjectURL(blob));else log('ERROR Could not encode PNG.')},'image/png');return}
  try{const url=canvas.toDataURL('image/png');const link=document.createElement('a');link.href=url;link.download=filename;link.click()}catch(error){log('ERROR PNG export failed: '+error.message)}
}

function runtimeDocsBase(){
  // Use the SAME raw repository root that already powers the 2D editor.
  // V8 converted this to jsDelivr for <script src>, which could leave mobile/local
  // copies waiting forever on the CDN. V9 fetches source text from repoBase instead.
  return base();
}
function runtimeUrl(path){return new URL(String(path).replace(/^\.\//,''),runtimeDocsBase()).href}
async function fetchRuntimeSource(path,timeoutMs=15000){
  const url=runtimeUrl(path);
  const controller=new AbortController();
  const timer=setTimeout(()=>controller.abort(),timeoutMs);
  try{
    const response=await fetch(url,{cache:'no-store',signal:controller.signal});
    if(!response.ok)throw new Error(`${response.status} ${response.statusText}`);
    return await response.text();
  }catch(error){
    if(error?.name==='AbortError')throw new Error(`Timed out fetching ${path}`);
    throw new Error(`Could not fetch ${path}: ${error?.message||error}`);
  }finally{clearTimeout(timer)}
}
function patchRuntimeSource(path,source){
  const docsExpr="new URL(window.__HobunjiRuntimeDocsBase || './', location.href)";
  let code=String(source||'');

  // Several hand/cloud helpers resolve assets from document.currentScript.
  // Because V9 executes fetched source directly rather than via a cross-origin
  // <script> tag, give those helpers the selected repository root explicitly.
  code=code.replace(
    /const selfUrl = document\.currentScript\?\.src \? new URL\(document\.currentScript\.src, location\.href\) : null;\s*const docsBase = selfUrl \? new URL\('\.\.\/', selfUrl\) : new URL\('\.\/', location\.href\);/g,
    `const selfUrl = null;\n  const docsBase = ${docsExpr};`
  );
  code=code.replace(
    /const selfUrl = document\.currentScript\?\.src \? new URL\(document\.currentScript\.src, location\.href\) : null;\s*const docsBase = selfUrl \? new URL\('\.\.\/', selfUrl\)\.href : new URL\('\.\/', location\.href\)\.href;/g,
    `const selfUrl = null;\n  const docsBase = ${docsExpr}.href;`
  );

  // procedural-leg-animation already has a repository-tool override; the line
  // remains untouched and reads __HobunjiProceduralFeetDocsBase.
  return `${code}\n//# sourceURL=${runtimeUrl(path)}`;
}
async function executeRuntimeScript(path,index,total){
  E.threeStatus.textContent=`Runtime ${index}/${total}: fetching ${path}…`;
  const source=await fetchRuntimeSource(path);
  E.threeStatus.textContent=`Runtime ${index}/${total}: starting ${path}…`;
  try{
    (0,eval)(patchRuntimeSource(path,source));
  }catch(error){
    throw new Error(`${path} threw during startup: ${error?.message||error}`);
  }
}
function installCanonicalCloudMaskScale(){
  // The live game wraps the ordinary destination-out portrait cloud mask with
  // this exact 0.60 Y-scale + bottom-anchor compensation. Keep that behavior
  // available even if the later arm-specific cloud helper elects not to wrap.
  const original=window.applyPortraitOpacityMask;
  if(typeof original!=='function'||original.__hobunjiCloudMaskYScaled)return;
  const wrapped=function(ctx,image,xform={}){
    const ax=Number(xform.ax)||0;
    const sy=Number(xform.sy??xform.scaleY)||1;
    const multiplier=.60;
    const bottomAnchorCompensation=sy*(1-multiplier)/2;
    return original(ctx,image,{
      ...xform,
      ax:ax-bottomAnchorCompensation,
      sy:sy*multiplier,
      scaleY:sy*multiplier
    });
  };
  wrapped.__hobunjiCloudMaskYScaled=true;
  window.applyPortraitOpacityMask=wrapped;
}
